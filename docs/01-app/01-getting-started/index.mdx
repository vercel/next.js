import { PrismaClient } from '@prisma/client';

const prisma = new PrismaClient();

async function main() {
  console.log('Bắt đầu dọn dẹp dữ liệu cũ...');
  await prisma.historicalScore.deleteMany();
  await prisma.majorMethod.deleteMany();
  await prisma.major.deleteMany();
  await prisma.university.deleteMany();

  console.log('Đang tạo dữ liệu Trường Đại học...');
  const ueh = await prisma.university.create({
    data: {
      code: 'KSA',
      name: 'Đại học Kinh tế TP.HCM (UEH)',
      logo: 'https://cdn.haitrieu.com/wp-content/uploads/2021/11/Logo-UEH.png',
    },
  });

  const hcmute = await prisma.university.create({
    data: {
      code: 'SPK',
      name: 'Đại học Sư phạm Kỹ thuật TP.HCM (HCMUTE)',
      logo: 'https://cdn.haitrieu.com/wp-content/uploads/2021/09/Logo-DH-Su-Pham-Ky-Thuat-TP-Ho-Chi-Minh.png',
    },
  });

  const bku = await prisma.university.create({
    data: {
      code: 'QSB',
      name: 'Đại học Bách Khoa - ĐHQG TP.HCM',
      logo: 'https://cdn.haitrieu.com/wp-content/uploads/2021/11/Logo-DH-Bach-Khoa-HCMUT.png',
    },
  });

  console.log('Đang tạo Ngành, Phương thức và Công thức tính điểm...');

  // 1. UEH - Marketing (Nhân hệ số 2 môn Toán, quy đổi IELTS)
  const uehMarketing = await prisma.major.create({
    data: {
      universityId: ueh.id,
      code: '7310105',
      name: 'Marketing',
      methods: {
        create: [
          {
            methodType: 'THPT',
            block: 'A00, A01, D01',
            formulaConfig: {
              type: 'THPT',
              multiplier: { Toan: 2, VatLy: 1, TiengAnh: 1 },
              divisor: 4,
              englishConversion: { IELTS: { '5.5': 8, '6.0': 9, '6.5': 10 } },
              priorityRule: 'STANDARD_30',
            },
            historicalScores: {
              create: [
                { year: 2025, score: 27.2 },
                { year: 2024, score: 27.0 },
                { year: 2023, score: 26.8 },
              ],
            },
          },
        ],
      },
    },
  });

  // 2. HCMUTE - Công nghệ thông tin (Tính tổng 3 môn bình thường)
  const hcmuteIT = await prisma.major.create({
    data: {
      universityId: hcmute.id,
      code: '7480201',
      name: 'Công nghệ thông tin',
      methods: {
        create: [
          {
            methodType: 'THPT',
            block: 'A00, A01, D01',
            formulaConfig: {
              type: 'THPT',
              multiplier: { Toan: 1, VatLy: 1, TiengAnh: 1 },
              priorityRule: 'STANDARD_30',
            },
            historicalScores: {
              create: [
                { year: 2025, score: 26.5 },
                { year: 2024, score: 26.1 },
                { year: 2023, score: 25.8 },
              ],
            },
          },
        ],
      },
    },
  });

  // 3. Bách Khoa (QSB) - Khoa học máy tính (Đánh giá năng lực)
  const bkuCS = await prisma.major.create({
    data: {
      universityId: bku.id,
      code: '7480101',
      name: 'Khoa học Máy tính',
      methods: {
        create: [
          {
            methodType: 'DGNL',
            block: 'DGNL',
            formulaConfig: {
              type: 'DGNL',
              priorityRule: 'DGNL_1200',
            },
            historicalScores: {
              create: [
                { year: 2025, score: 1050 },
                { year: 2024, score: 1035 },
                { year: 2023, score: 1020 },
              ],
            },
          },
        ],
      },
    },
  });

  console.log('Seed dữ liệu thành công! Đã thêm dữ liệu mẫu cho KSA, SPK và QSB.');
}

main()
  .catch((e) => {
    console.error('Lỗi khi seed dữ liệu:', e);
    process.exit(1);
  })
  .finally(async () => {
    await prisma.$disconnect();
  });
