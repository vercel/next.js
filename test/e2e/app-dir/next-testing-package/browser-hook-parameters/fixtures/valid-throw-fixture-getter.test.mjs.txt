import { test, expect, beforeEach, afterEach, afterAll } from 'vitest'
const trace = [], originalError = new Error('authored hook parameter original error')
let bodyCalls = 0
function emit(value) { console.log('BROWSER_NATIVE_FIXTURE:' + JSON.stringify(value)) }
function throwOriginal() {
  trace.push('parameter:throw')
  emit({ phase: 'throw', trace: [...trace], originalName: originalError.name, originalMessage: originalError.message, originalOwnKeys: Object.keys(originalError) })
  throw originalError
}
function observedErrors(task) {
  const errors = task && task.result && task.result.errors || []
  const facts = []
  for (const error of errors) facts.push({ sameOriginal: error === originalError, name: error.name, message: error.message, ownKeys: Object.keys(error).sort() })
  return facts
}
afterEach(function namedFailureObserver({ task }) {
  trace.push('afterEach:body')
  expect(bodyCalls).toBe(0)
  emit({ phase: 'afterEach', trace: [...trace], bodyCalls, errors: observedErrors(task) })
})
afterAll(function namedFinalObserver() {
  trace.push('afterAll:body')
  emit({ phase: 'afterAll', trace: [...trace], bodyCalls })
})
const danger = { get value() { trace.push('fixture:getter'); return throwOriginal() } }
const fixtureTest = test.extend({ danger })
beforeEach(function namedFixtureGetter({ danger: { value } }) {
  bodyCalls++
  trace.push('beforeEach:body')
})
fixtureTest('retains the original error thrown by a genuine fixture getter during hook parameter initialization', function namedUnstartedFixtureCase() {
  bodyCalls++
  trace.push('test:body')
})
