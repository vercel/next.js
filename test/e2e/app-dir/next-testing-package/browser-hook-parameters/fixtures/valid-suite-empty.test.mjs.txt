import { test, expect, beforeAll, afterAll, beforeEach, afterEach, beforeEach as setup } from 'vitest'
const trace = []
function emit(value) { console.log('BROWSER_NATIVE_FIXTURE:' + JSON.stringify(value)) }
function record(phase, fn, args, target, receiver, value, rest) {
  trace.push(phase + ':body')
  emit({ phase, trace: [...trace], function: { name: fn.name, length: fn.length, ownPrototype: Object.hasOwn(fn, 'prototype'), prototypeConstructor: fn.prototype.constructor === fn }, arguments: { length: args.length, firstType: typeof args[0], firstHasTask: !!args[0] && 'task' in args[0], restKeys: rest === undefined ? null : Object.keys(rest).sort() }, newTargetUndefined: target === undefined, receiverUndefined: receiver === undefined, initializedValue: value })
}
beforeAll(function namedPlainBeforeAll() {
  record('plainBeforeAll', namedPlainBeforeAll, arguments, new.target, this, undefined, undefined)
})
beforeAll(function namedEmptyBeforeAll({}) {
  record('emptyBeforeAll', namedEmptyBeforeAll, arguments, new.target, this, undefined, undefined)
})
afterAll(function namedEmptyAfterAll({}) {
  record('emptyAfterAll', namedEmptyAfterAll, arguments, new.target, this, undefined, undefined)
})
afterAll(function namedPlainAfterAll() {
  record('plainAfterAll', namedPlainAfterAll, arguments, new.target, this, undefined, undefined)
})
test('uses original empty-object parameters in direct synchronous suite hooks', function namedEmptyCase() {
  trace.push('test:body')
  expect(trace).toEqual(['plainBeforeAll:body', 'emptyBeforeAll:body', 'test:body'])
  emit({ phase: 'test', trace: [...trace] })
})
