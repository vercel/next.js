import { test, expect, beforeAll, afterAll } from 'vitest'
const trace = []
function emit(value) {
  console.log('BROWSER_NATIVE_FIXTURE:' + JSON.stringify(value))
}
function record(phase, fn, args, target, receiver, value, rest) {
  trace.push(phase + ':body')
  expect(fn.name).toBe('named' + phase[0].toUpperCase() + phase.slice(1))
  expect(target).toBeUndefined()
  expect(receiver).toBeUndefined()
  if (phase.endsWith('Each')) {
    expect(args).toHaveLength(1)
    expect(args[0].task.context).toBe(args[0])
    expect(rest.task).toBe(args[0].task)
    expect(rest.signal).toBe(args[0].signal)
    expect(value).toBe(phase + '-value')
  } else {
    expect(args).toHaveLength(2)
    expect(typeof args[0]).toBe('object')
    expect('task' in args[0]).toBe(false)
  }
  emit({ phase })
}
beforeAll(function namedPlainBeforeAll() {
  record(
    'plainBeforeAll',
    namedPlainBeforeAll,
    arguments,
    new.target,
    this,
    undefined,
    undefined
  )
})
beforeAll(function namedEmptyBeforeAll({}) {
  record(
    'emptyBeforeAll',
    namedEmptyBeforeAll,
    arguments,
    new.target,
    this,
    undefined,
    undefined
  )
})
afterAll(function namedEmptyAfterAll({}) {
  record(
    'emptyAfterAll',
    namedEmptyAfterAll,
    arguments,
    new.target,
    this,
    undefined,
    undefined
  )
})
afterAll(function namedPlainAfterAll() {
  record(
    'plainAfterAll',
    namedPlainAfterAll,
    arguments,
    new.target,
    this,
    undefined,
    undefined
  )
})
test('uses original empty-object parameters in direct synchronous suite hooks', function namedEmptyCase() {
  trace.push('test:body')
  expect(trace).toEqual([
    'plainBeforeAll:body',
    'emptyBeforeAll:body',
    'test:body',
  ])
  emit({ phase: 'test' })
})
