import Page from '../../late-redirect/[...slug]/page'
import { headers, cookies } from 'next/headers'
import { tenant as rootTenant } from 'next/root-params'
import { after } from 'next/server'
import { ownerAfter, afterErrors } from './state'

export const dynamic = 'force-dynamic'
export const runtime = 'nodejs'

export default async function StreamOwnerPage(props) {
  const query = await props.searchParams
  const requestHeaders = await headers()
  const requestCookies = await cookies()
  const request = requestHeaders.get('x-route-request')
  const barrier = requestHeaders.get('x-after-barrier')
  const row = { request, root: await rootTenant(), header: requestHeaders.get('x-route-proof'), cookie: requestCookies.get('route-session')?.value ?? null, outcome: query.after === 'reject' ? 'rejected' : 'fulfilled' }
  after(async () => {
    if (barrier) {
      const response = await fetch(barrier, { cache: 'no-store' })
      if (!response.ok) throw new Error('Owned after barrier rejected')
    }
    ownerAfter.push(row)
    console.log('STREAM_OWNER_AFTER ' + JSON.stringify(row))
    if (query.after === 'reject') {
      const error = new Error('Owned real AppPage after failure')
      afterErrors.set(request, error)
      throw error
    }
  })
  return <Page {...props} />
}
