import { afterAll, expect, test, vi, vitest } from 'vitest'
const key = Symbol.for('next-test-reset-utility-identity-counter')
const original = Object.getOwnPropertyDescriptor(globalThis, key)
const collectionReset = vi.resetModules
let retainedReset
const emit = (label, value) => console.log('RESET_UTILITY_ROW ' + JSON.stringify({ label, value }))
function describe(fn) {
  let constructible
  try { Reflect.construct(fn, []); constructible = true } catch { constructible = false }
  const name = Object.getOwnPropertyDescriptor(fn, 'name')
  const length = Object.getOwnPropertyDescriptor(fn, 'length')
  return { name: fn.name, length: fn.length, ownNames: Object.getOwnPropertyNames(fn), hasPrototype: Object.hasOwn(fn, 'prototype'), constructible, nameDescriptor: { value: name.value, writable: name.writable, enumerable: name.enumerable, configurable: name.configurable }, lengthDescriptor: { value: length.value, writable: length.writable, enumerable: length.enumerable, configurable: length.configurable } }
}
afterAll(() => { if (original) Object.defineProperty(globalThis, key, original); else Reflect.deleteProperty(globalThis, key) })
test('resetModules retains one ordinary method identity and its real descriptors', () => {
  retainedReset = vi.resetModules
  const own = Object.getOwnPropertyDescriptor(vi, 'resetModules')
  const property = own ? { kind: 'value' in own ? 'data' : 'accessor', sameValue: own.value === retainedReset, writable: own.writable, enumerable: own.enumerable, configurable: own.configurable } : null
  const value = { property, repeat: retainedReset === vi.resetModules, collection: retainedReset === collectionReset, namespaceAlias: retainedReset === vitest.resetModules, shape: describe(retainedReset), extractedReturn: retainedReset() === vi, foreignReceiverReturn: retainedReset.call({}) === vi, boundShape: describe(retainedReset.bind({})), boundReturn: retainedReset.bind({})() === vi }
  emit('identity', value)
  expect(value.property).toEqual({ kind: 'data', sameValue: true, writable: true, enumerable: true, configurable: true })
  expect(value.repeat).toBe(true)
  expect(value.collection).toBe(true)
  expect(value.namespaceAlias).toBe(true)
  expect(value.shape.name).toBe('resetModules')
  expect(value.shape.length).toBe(0)
  expect(value.shape.hasPrototype).toBe(false)
  expect(value.shape.constructible).toBe(false)
  expect(value.extractedReturn).toBe(true)
  expect(value.foreignReceiverReturn).toBe(true)
  expect(value.boundReturn).toBe(true)
})
test('direct reset invalidates the actual imported module cache', async () => {
  const first = await import('./counter.mjs')
  const same = await import('./counter.mjs')
  const resetReturn = vi.resetModules() === vi
  const second = await import('./counter.mjs')
  const value = { first: first.ordinal, same: same === first, second: second.ordinal, fresh: second !== first, resetReturn }
  emit('direct-cache', value)
  expect(value).toEqual({ first: 1, same: true, second: 2, fresh: true, resetReturn: true })
})
test('an alias retained from a completed case resets the current real cache', async () => {
  const before = await import('./counter.mjs')
  const retainedReturn = retainedReset() === vi
  const after = await import('./counter.mjs')
  const value = { before: before.ordinal, after: after.ordinal, fresh: before !== after, retainedReturn, stillCurrentMethod: retainedReset === vi.resetModules }
  emit('retained-cache', value)
  expect(value).toEqual({ before: 2, after: 3, fresh: true, retainedReturn: true, stillCurrentMethod: true })
})
test('the collection alias resolves the real active runtime when later invoked', async () => {
  const before = await import('./counter.mjs')
  const collectionReturn = collectionReset() === vi
  const after = await import('./counter.mjs')
  const value = { before: before.ordinal, after: after.ordinal, fresh: before !== after, collectionReturn, stillCurrentMethod: collectionReset === vi.resetModules }
  emit('collection-cache', value)
  expect(value).toEqual({ before: 3, after: 4, fresh: true, collectionReturn: true, stillCurrentMethod: true })
})
test('the actual resetModules property preserves replacement, deletion and getter receivers', () => {
  const descriptor = Object.getOwnPropertyDescriptor(vi, 'resetModules')
  const receiver = { marker: 'foreign receiver' }
  const replacement = function () { return this }
  let value
  try {
    vi.resetModules = replacement
    const replacementIdentity = vi.resetModules === replacement
    const foreignReceiver = Reflect.apply(vi.resetModules, receiver, []) === receiver
    const retainedAliasReturn = collectionReset() === vi
    const deleted = Reflect.deleteProperty(vi, 'resetModules')
    const absent = vi.resetModules === undefined
    let getterReceiver
    const expected = descriptor?.value ?? collectionReset
    Object.defineProperty(vi, 'resetModules', { configurable: true, enumerable: false, get() { getterReceiver = this; return expected } })
    const getterIdentity = vi.resetModules === expected
    value = { replacementIdentity, foreignReceiver, retainedAliasReturn, deleted, absent, getterIdentity, getterReceiver: getterReceiver === vi }
  } finally {
    if (descriptor) Object.defineProperty(vi, 'resetModules', descriptor)
    else Reflect.deleteProperty(vi, 'resetModules')
  }
  emit('property', value)
  expect(value).toEqual({ replacementIdentity: true, foreignReceiver: true, retainedAliasReturn: true, deleted: true, absent: true, getterIdentity: true, getterReceiver: true })
})
