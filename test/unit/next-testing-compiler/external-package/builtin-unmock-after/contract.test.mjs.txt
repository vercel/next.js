import { test, expect, vi } from 'vitest'
vi.mock('path', () => {
  globalThis.ws2CancelledCalls = (globalThis.ws2CancelledCalls ?? 0) + 1
  return {
    join: () => 'cancelled',
  }
})
vi.unmock('node:path')
vi.mock('node:os', () => {
  globalThis.ws2KeptCalls = (globalThis.ws2KeptCalls ?? 0) + 1
  return {
    platform: () => 'kept',
  }
})
import * as path from 'path'
import * as prefixed from 'node:path'
import * as os from 'os'
test('builtin-unmock-after', async () => {
  const actual = await vi.importActual('node:path')
  const dynamic = await import('node:path')
  const descriptor = Object.getOwnPropertyDescriptor(path, 'join')
  {
    const observed = {
      value: path.join('alpha', 'beta'),
      prefixedValue: prefixed.join('alpha', 'beta'),
      dynamicValue: dynamic.join('alpha', 'beta'),
      actual: actual.posix.join('alpha', 'beta'),
      identity: path === prefixed,
      dynamicIdentity: path === dynamic,
      actualIdentity: path === actual,
      cancelledCalls: globalThis.ws2CancelledCalls ?? 0,
      keptCalls: globalThis.ws2KeptCalls ?? 0,
      os: os.platform(),
    }
    expect(observed.value).toBe('alpha/beta')
    expect(observed.prefixedValue).toBe('alpha/beta')
    expect(observed.dynamicValue).toBe('alpha/beta')
    expect(observed.actual).toBe('alpha/beta')
    expect(observed.identity).toBe(true)
    expect(observed.dynamicIdentity).toBe(true)
    expect(observed.actualIdentity).toBe(true)
    expect(observed.cancelledCalls).toBe(0)
    expect(observed.keptCalls).toBe(1)
    expect(observed.os).toBe('kept')
  }
  expect(actual.posix.join('alpha', 'beta')).toBe('alpha/beta')
})
