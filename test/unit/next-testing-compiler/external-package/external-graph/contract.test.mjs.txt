import { test, expect } from 'vitest'
import a from 'ws2-external-a'
import b from 'ws2-external-b'
test('preserves native external dependency diamonds cycles and package data', () => {
  const row = {
    diamond: a.c === a.b.c,
    cycle: a.c.a === a.object,
    rootAndDependency: a.b === b,
    data: a.c.data,
  }
  {
    const observed = row
    expect(observed.diamond).toBe(true)
    expect(observed.cycle).toBe(true)
    expect(observed.rootAndDependency).toBe(true)
    expect(observed.data).toBe('captured-package-data\n')
  }
  expect(row).toEqual({
    diamond: true,
    cycle: true,
    rootAndDependency: true,
    data: 'captured-package-data\n',
  })
})
