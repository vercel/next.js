 vi.doMock("node:fs",()=>{globalThis.ws2StaticRuntimeRuntimeFactories=(globalThis.ws2StaticRuntimeRuntimeFactories??0)+1;return {value:73}})

 const row={staticFactories:globalThis.ws2StaticRuntimeStaticFactories??0,runtimeFactories:globalThis.ws2StaticRuntimeRuntimeFactories??0,originalEvaluations:null}
 console.log('WS2_EXTERNAL_STATIC_RUNTIME_UNUSED:'+JSON.stringify(row))
 expect(row.staticFactories).toBe(0)
 expect(row.runtimeFactories).toBe(0)
 
})
