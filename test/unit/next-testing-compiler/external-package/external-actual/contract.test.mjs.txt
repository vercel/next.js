import {test,expect,vi} from 'vitest'
vi.mock('ws2-genuine-external-original',async(importOriginal)=>{const actual=await importOriginal();return {value:73,original:actual}})
vi.mock('ws2-genuine-external-esm',async(importOriginal)=>{const actual=await importOriginal();return {value:74,original:actual}})
import * as common from 'ws2-genuine-external-original'
import * as esm from 'ws2-genuine-external-esm'
import * as conditional from 'ws2-external-conditions'
function row(value){console.log('WS2_EXTERNAL_ACTUAL:'+JSON.stringify(value))}
test('loads a mocked real CJS package through literal importActual',async()=>{const first=await vi.importActual('ws2-genuine-external-original');const second=await vi.importActual('ws2-genuine-external-original');row({name:'actual-cjs',mockValue:common.value,value:first.value,defaultValue:first.default.value,same:first===second,lazyOriginal:first===common.original,objectIdentity:first.object===first.default.object,tag:first[Symbol.toStringTag],keys:Object.keys(first),prototypeNull:Object.getPrototypeOf(first)===null,valueDescriptor:Object.getOwnPropertyDescriptor(first,'value')});expect(first.value).toBe(19);expect(first).toBe(common.original)})
test('loads a mocked real ESM package through literal importActual',async()=>{const first=await vi.importActual('ws2-genuine-external-esm');const second=await vi.importActual('ws2-genuine-external-esm');row({name:'actual-esm',mockValue:esm.value,value:first.value,defaultValue:first.default.value,same:first===second,lazyOriginal:first===esm.original,tag:first[Symbol.toStringTag],keys:Object.keys(first),prototypeNull:Object.getPrototypeOf(first)===null});expect(first.value).toBe(37);expect(first).toBe(esm.original)})
test('shares the actual native conditional import target with ordinary imports',async()=>{const first=await vi.importActual('ws2-external-conditions');const second=await vi.importActual('ws2-external-conditions');const dynamic=await import('ws2-external-conditions');row({name:'actual-conditional',branch:first.branch,object:first.object,default:first.default,same:first===second,staticIdentity:first===conditional,dynamicIdentity:first===dynamic,tag:first[Symbol.toStringTag],keys:Object.keys(first)});expect(first.branch).toBe('native-import');expect(first).toBe(conditional)})
