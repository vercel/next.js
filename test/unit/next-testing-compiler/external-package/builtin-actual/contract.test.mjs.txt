import { test, expect, vi } from 'vitest'
vi.mock('node:path', async (importOriginal) => {
  const original = await importOriginal()
  return {
    join: () => 'mock-join',
    original,
  }
})
import * as path from 'node:path'
import * as fs from 'fs'
import * as os from 'node:os'
test('shares literal builtin actual namespaces with native lazy originals', async () => {
  const bare = await vi.importActual('path')
  const prefixed = await vi.importActual('node:path')
  {
    const observed = {
      mockValue: path.join(),
      actualValue: bare.posix.join('a', 'b'),
      aliases: bare === prefixed,
      lazyOriginal: bare === path.original,
      defaultFunction: bare.default.join === bare.join,
    }
    expect(observed.mockValue).toBe('mock-join')
    expect(observed.actualValue).toBe('a/b')
    expect(observed.aliases).toBe(true)
    expect(observed.lazyOriginal).toBe(true)
    expect(observed.defaultFunction).toBe(true)
  }
  expect(bare).toBe(path.original)
})
test('shares unmocked bare builtin static dynamic and actual namespaces', async () => {
  const bare = await vi.importActual('fs')
  const prefixed = await vi.importActual('node:fs')
  const dynamic = await import('node:fs')
  {
    const observed = {
      aliases: bare === prefixed,
      staticIdentity: bare === fs,
      dynamicIdentity: bare === dynamic,
      defaultFunction: bare.default.readFileSync === bare.readFileSync,
    }
    expect(observed.aliases).toBe(true)
    expect(observed.staticIdentity).toBe(true)
    expect(observed.dynamicIdentity).toBe(true)
    expect(observed.defaultFunction).toBe(true)
  }
  expect(bare).toBe(fs)
})
test('shares unmocked prefixed builtin namespaces with literal actual helpers', async () => {
  const bare = await vi.importActual('os')

    }

  }
  expect(bare).toBe(os)
