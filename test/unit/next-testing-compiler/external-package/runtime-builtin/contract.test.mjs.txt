import { test, expect, vi } from 'vitest'
import { types as utilTypes } from 'node:util'
import * as staticOriginal from 'node:os'
function shape(namespace) {
  return { nativeBrand: utilTypes.isModuleNamespaceObject(namespace), keys: Reflect.ownKeys(namespace).map((key) => typeof key === 'symbol' ? String(key) : key),
    prototypeNull: Object.getPrototypeOf(namespace) === null, tag: namespace[Symbol.toStringTag] ?? null,
    extensible: Object.isExtensible(namespace), sealed: Object.isSealed(namespace), frozen: Object.isFrozen(namespace),
    descriptors: Reflect.ownKeys(namespace).map((key) => {
      const d = Object.getOwnPropertyDescriptor(namespace, key)
      return { key: typeof key === 'symbol' ? String(key) : key, valueType: typeof d.value, get: typeof d.get, set: typeof d.set,
        writable: d.writable ?? null, enumerable: d.enumerable, configurable: d.configurable }
    }) }
}
test('runtime native builtin registration unregistration and reset identity', async () => {
  const before = await import('node:os')
  const beforeAlias = await import('os')
  let calls = 0
  let value = 73
  vi.doMock('node:os', async (importOriginal) => {
    calls++
    const original = await importOriginal()
    return { value, original: typeof original.platform, sameOriginal: original === before }
  })
  const first = await import('os')
  value = 81
  const second = await import('node:os')
  vi.resetModules()
  const afterMockReset = await import('os')
  vi.doUnmock('os')
  const released = await import('node:os')
  vi.resetModules()
  const afterOriginalReset = await import('os')
  vi.doMock('os', () => ({ value: 91 }))
  const replacement = await import('node:os')
  vi.doUnmock('node:os')
  const final = await import('os')
  const row = { before: typeof before.platform, first: first.value, second: second.value,
    mockIdentity: first === second, original: first.original, sameOriginal: first.sameOriginal,
    afterMockReset: afterMockReset.value, mockResetIdentity: first === afterMockReset, calls,
    released: typeof released.platform, releasedIdentity: released === before,
    originalResetIdentity: afterOriginalReset === before,
    replacement: replacement.value, replacementIdentity: replacement === first,
    finalIdentity: final === before, heldOriginal: typeof before.platform, heldMock: first.value, staticBeforeIdentity: staticOriginal === before, staticReleasedIdentity: staticOriginal === released, staticBrand: utilTypes.isModuleNamespaceObject(staticOriginal), staticShape: shape(staticOriginal), beforeAliasIdentity: beforeAlias === before,
    beforeShape: shape(before), releasedShape: shape(released), afterOriginalResetShape: shape(afterOriginalReset),
    defaultIdentity: before.default === released.default, memberIdentity: before.platform === released.platform }
  console.log('WS2_EXTERNAL_NATIVE_RUNTIME:' + JSON.stringify(row))
  expect(first.value).toBe(73)
  expect(replacement.value).toBe(91)
})

test('runtime native builtin registration without reset', async () => {
  const before = await import('node:os')
  let calls = 0
  vi.doMock('node:os', async (importOriginal) => {
    calls++
    const actual = await importOriginal()
    return { value: 73, sameOriginal: actual === before }
  })
  const mocked = await import('os')
  vi.doUnmock('os')
  const released = await import('node:os')
  vi.doMock('os', () => ({ value: 91 }))
  const replaced = await import('node:os')
  vi.doUnmock('node:os')
  console.log('WS2_EXTERNAL_NATIVE_RUNTIME_WITHOUT_RESET:' + JSON.stringify({
    value: mocked.value, sameOriginal: mocked.sameOriginal, calls,
    releasedIdentity: before === released, replaced: replaced.value,
    replacedIdentity: mocked === replaced, held: mocked.value,
    finalIdentity: (await import('os')) === before
  }))
  expect(mocked.value).toBe(73)
  expect(replaced.value).toBe(91)
})
test('runtime native builtin caches a failed factory across reset and can unregister', async () => {
  const original = await import('node:os')
  const failure = new Error('ws2 native runtime factory failure')
  let calls = 0
  vi.doMock('node:os', () => { calls++; throw failure })
  const first = await import('os').then(() => null, (error) => error)
  const second = await import('node:os').then(() => null, (error) => error)
  vi.resetModules()
  const third = await import('os').then(() => null, (error) => error)
  vi.doUnmock('os')
  const released = await import('node:os')
  console.log('WS2_EXTERNAL_NATIVE_RUNTIME_FAILURE:' + JSON.stringify({
    calls, firstIsFailure: first === failure, sameSecond: first === second, sameAfterReset: first === third,
    error: { name: first?.name ?? null, message: first?.message ?? null,
      causeName: first?.cause?.name ?? null, causeMessage: first?.cause?.message ?? null,
      causeIsFailure: first?.cause === failure },
    releasedIdentity: original === released, releasedShape: shape(released)
  }))
  expect(first).not.toBe(null)
})

test('runtime native unused registration retains compiler IDs without evaluating originals', () => {
  vi.doMock('ws2-native-runtime-unused', () => {
    globalThis.ws2UnusedNativeFactories = (globalThis.ws2UnusedNativeFactories ?? 0) + 1
    return { value: 1 }
  })
  vi.doUnmock('ws2-native-runtime-unused')
  const row = { evaluations: globalThis.ws2UnusedNativeEvaluations ?? 0, factories: globalThis.ws2UnusedNativeFactories ?? 0 }
  console.log('WS2_EXTERNAL_NATIVE_RUNTIME_UNUSED:' + JSON.stringify(row))
  expect(row).toEqual({ evaluations: 0, factories: 0 })
})

test('runtime native builtin pending RESOLVE imports and later cache', async () => {
  const heldOriginal = await import('node:os')
  let release
  const held = new Promise((resolve, reject) => { release = resolve })
  const failure = new Error('ws2 pending native failure')
  let calls = 0
  vi.doMock('node:os', async () => {
    const index = ++calls
    await held
    return { value: 73, index }
  })
  const firstPending = import('os').then(value => ({ value }), error => ({ error }))
  const secondPending = import('node:os').then(value => ({ value }), error => ({ error }))
  let twoFactoriesEntered = false
  try { await vi.waitFor(() => expect(calls).toBe(2), { timeout: 1000, interval: 10 }); twoFactoriesEntered = true } catch {}
  const callsWhileHeld = calls
  release(undefined)
  const first = await firstPending
  const second = await secondPending
  const later = await import('node:os').then(value => ({ value }), error => ({ error }))
  vi.resetModules()
  const afterReset = await import('os').then(value => ({ value }), error => ({ error }))
  vi.doUnmock('os')
  const errorShape = value => value ? { name: value.name, message: value.message, cause: value.cause === failure } : null
  console.log('WS2_EXTERNAL_NATIVE_RUNTIME_PENDING_RESOLVE:' + JSON.stringify({
    twoFactoriesEntered, callsWhileHeld, calls, secondOriginalIdentity: second.value === heldOriginal, secondOriginalBrand: second.value ? utilTypes.isModuleNamespaceObject(second.value) : null, secondOriginalDefault: second.value?.default === heldOriginal.default, secondOriginalShape: second.value ? shape(second.value) : null, firstIndex: first.value?.index ?? null,
    secondIndex: second.value?.index ?? null, sameValues: first.value === second.value,
    sameErrors: first.error === second.error, firstError: errorShape(first.error), secondError: errorShape(second.error),
    laterIndex: later.value?.index ?? null, laterIsFirst: later.value === first.value,
    laterIsSecond: later.value === second.value, laterError: errorShape(later.error),
    laterErrorIsFirst: later.error === first.error,
    resetIndex: afterReset.value?.index ?? null, resetIsLater: afterReset.value === later.value,
    resetError: errorShape(afterReset.error), resetErrorIsLater: afterReset.error === later.error
  }))
  expect(callsWhileHeld).toBeGreaterThanOrEqual(1)
})
