test('throw', async () => {
const name = 'throw';
    const originalDefine = Object.defineProperty;
    const failure = new Error('authored body definition failure');
    const trace = [];
    Object.defineProperty = function(target, key, descriptor) {
      if (key === '__esModule' && typeof target === 'function') {
        trace.push({ event: 'flag-definition', keys: Object.keys(descriptor), bodyCount: globalThis.__bodyDefinitions?.[name] ?? 0 });
        if (name === 'throw') throw failure;
        return target;
      }
      if (key === Symbol.toStringTag && descriptor.value === 'Module' && Object.getPrototypeOf(target) === null) {
        trace.push({ event: 'namespace-tag', bodyCount: globalThis.__bodyDefinitions?.[name] ?? 0, keys: Reflect.ownKeys(target).map(String) });
      }
      return Reflect.apply(originalDefine, this, [target, key, descriptor]);
    };
    let mocked, error;
