import { test, expect, vi } from 'vitest'
import { types } from 'node:util'
import * as heldStatic from 'ws2-genuine-external-original'
vi.mock('ws2-genuine-external-original', () => ({
  value: 5,
}))
test('native CJS static and runtime registrations retain authored ordering', async () => {
  const before = await import('ws2-genuine-external-original')
  let calls = 0
  let actual
  vi.doMock('ws2-genuine-external-original', async (original) => {
    calls++
    actual = await original()
    return {
      value: 73,
      actualValue: actual.value ?? null,
      actualReadFile: typeof actual.readFile,
      sameStatic: actual === heldStatic,
    }
  })
  const mocked = await import('ws2-genuine-external-original')
  vi.resetModules()
  const afterMockReset = await import('ws2-genuine-external-original')
  vi.doUnmock('ws2-genuine-external-original')
  const released = await import('ws2-genuine-external-original')
  vi.doMock('ws2-genuine-external-original', () => ({
    value: 91,
  }))
  const replacement = await import('ws2-genuine-external-original')
  vi.resetModules()
  vi.doUnmock('ws2-genuine-external-original')
  const final = await import('ws2-genuine-external-original')
  {
    const observed = {
      calls,
      values: [before.value, mocked.value, replacement.value, heldStatic.value],
      actualValue: mocked.actualValue,
      actualReadFile: mocked.actualReadFile,
      actualBrand: types.isModuleNamespaceObject(actual),
      sameActualStatic: mocked.sameStatic,
      staticBefore: heldStatic === before,
      mockReset: mocked === afterMockReset,
      releasedValue: released.value ?? null,
      releasedReadFile: typeof released.readFile,
      releasedActual: released === actual,
      finalReleased: final === released,
      heldMockValue: mocked.value,
      actualDefaultReleased: actual.default === released.default,
      actualFnReleased: actual.fn === released.fn,
      releasedDefaultFinal: released.default === final.default,
      releasedFnFinal: released.fn === final.fn,
    }
    expect(observed.calls).toBe(1)
    expect(observed.values).toEqual([5, 73, 91, 5])
    expect(observed.actualValue).toBe(19)
    expect(observed.actualReadFile).toBe('undefined')
    expect(observed.actualBrand).toBe(false)
    expect(observed.sameActualStatic).toBe(false)
    expect(observed.staticBefore).toBe(true)
    expect(observed.mockReset).toBe(true)
    expect(observed.releasedValue).toBe(19)
    expect(observed.releasedReadFile).toBe('undefined')
    expect(observed.releasedActual).toBe(false)
    expect(observed.finalReleased).toBe(false)
    expect(observed.heldMockValue).toBe(73)
    expect(observed.actualDefaultReleased).toBe(true)
    expect(observed.actualFnReleased).toBe(true)
    expect(observed.releasedDefaultFinal).toBe(true)
    expect(observed.releasedFnFinal).toBe(true)
  }
  expect(mocked.value).toBe(73)
  expect(replacement.value).toBe(91)
})
