import { test, expect, vi } from 'vitest'
import * as heldStatic from 'ws2-genuine-external-original'
vi.mock('ws2-genuine-external-original', () => ({
  value: 5,
}))
test('native cjs static and runtime pending-resolve', async () => {
  const before = await import('ws2-genuine-external-original')
  let release
  const held = new Promise((resolve, reject) => {
    release = resolve
  })
  const failure = new Error('static runtime native pending failure')
  let calls = 0
  let actual
  vi.doMock('ws2-genuine-external-original', async (original) => {
    const index = ++calls
    actual = await original()
    await held
    return {
      value: 73,
      index,
    }
  })
  const firstPending = import('ws2-genuine-external-original').then(
    (value) => ({
      value,
    }),
    (error) => ({
      error,
    })
  )
  await vi.waitFor(() => expect(actual).toBeDefined())
  let secondSettled = false
  const secondPending = import('ws2-genuine-external-original').then(
    (value) => {
      secondSettled = true
      return {
        value,
      }
    },
    (error) => {
      secondSettled = true
      return {
        error,
      }
    }
  )
  let settledBeforeRelease = false
  try {
    await vi.waitFor(() => expect(secondSettled).toBe(true), {
      timeout: 250,
      interval: 10,
    })
    settledBeforeRelease = true
  } catch {}
  const callsWhileHeld = calls
  const heldActual = actual
  release(undefined)
  const first = await firstPending
  const second = await secondPending
  const later = await import('ws2-genuine-external-original').then(
    (value) => ({
      value,
    }),
    (error) => ({
      error,
    })
  )
  vi.resetModules()
  const afterReset = await import('ws2-genuine-external-original').then(
    (value) => ({
      value,
    }),
    (error) => ({
      error,
    })
  )
  vi.doUnmock('ws2-genuine-external-original')
  const released = await import('ws2-genuine-external-original')
  const errorShape = (error) =>
    error
      ? {
          name: error.name,
          message: error.message,
          causeIdentity: error.cause === failure,
          causeName: error.cause?.name ?? null,
          causeMessage: error.cause?.message ?? null,
          rawIdentity: error === failure,
        }
      : null
  const row = {
    calls,
    callsWhileHeld,
    settledBeforeRelease,
    beforeStatic: before === heldStatic,
    staticValue: heldStatic.value,
    secondOriginal: second.value === heldActual,
    secondStatic: second.value === heldStatic,
    values: [first, second, later, afterReset].map(
      (item) => item.value?.value ?? null
    ),
    mockIdentity: [
      first.value === later.value,
      later.value === afterReset.value,
    ],
    errorIdentity: [
      first.error === later.error,
      later.error === afterReset.error,
    ],
    errors: [first, second, later, afterReset].map((item) =>
      errorShape(item.error)
    ),
    releasedActual: released === actual,
    releasedHeldActual: released === heldActual,
    releasedValue: released.value ?? null,
    heldOriginalDefault: heldActual.default === released.default,
    heldOriginalFunction: heldActual.fn === released.fn,
  }
  {
    const observed = row
    expect(observed.calls).toBe(1)
    expect(observed.callsWhileHeld).toBe(1)
    expect(observed.settledBeforeRelease).toBe(true)
    expect(observed.beforeStatic).toBe(true)
    expect(observed.staticValue).toBe(5)
    expect(observed.secondOriginal).toBe(true)
    expect(observed.secondStatic).toBe(false)
    expect(observed.values).toEqual([73, 19, 73, 73])
    expect(observed.mockIdentity).toEqual([true, true])
    expect(observed.errorIdentity).toEqual([true, true])
    expect(observed.errors).toEqual([null, null, null, null])
    expect(observed.releasedActual).toBe(false)
    expect(observed.releasedHeldActual).toBe(false)
    expect(observed.releasedValue).toBe(19)
    expect(observed.heldOriginalDefault).toBe(true)
    expect(observed.heldOriginalFunction).toBe(true)
  }
  expect(callsWhileHeld).toBe(1)
  expect(heldStatic.value).toBe(5)
})
