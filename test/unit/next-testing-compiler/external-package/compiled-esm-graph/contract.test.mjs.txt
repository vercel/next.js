test('graph runtime failures retry with original graph identity and reset cache boundaries',async()=>{
 const cause=new Error('graph factory failed');let calls=0;const originals=[];let failing=true
 vi.doMock('./retry/target.mjs',async importOriginal=>{calls++;const actual=await importOriginal();originals.push(actual);if(failing)throw cause;return {token:actual.token}})
 const first=await import('./retry/target.mjs').catch(error=>error);const second=await import('./retry/target.mjs').catch(error=>error);vi.resetModules();const third=await import('./retry/target.mjs').catch(error=>error);vi.doUnmock('./retry/target.mjs');const released=await import('./retry/target.mjs')
 console.log('WS2_EXTERNAL_GRAPH_RETRY:'+JSON.stringify({calls,errors:[errorRow(first,cause),errorRow(second,cause),errorRow(third,cause)],errorIdentity:[first===second,second===third],originalIdentity:[originals[0]===originals[1],originals[1]===originals[2],originals[2]===released],evaluations:[...originals.map(original=>original.evaluation),released.evaluation],counts:counts('retry')}));expect(calls).toBe(3);expect(third.cause).toBe(cause)
})
