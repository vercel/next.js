import {test,expect,vi} from 'vitest'
vi.mock('./dep.mjs',()=>({value:73}))
import * as mocked from './dep.mjs'
import * as fs from 'node:fs'
test('combines compiled mock helpers with an ordinary native builtin namespace',async()=>{const helper=await vi.importMock('./dep.mjs');const actual=await vi.importActual('./dep.mjs');const dynamic=await import('fs');const descriptor=Object.getOwnPropertyDescriptor(fs,'readFileSync');console.log('WS2_EXTERNAL_COMPILED_BUILTIN:'+JSON.stringify({value:mocked.value,helperValue:helper.value,helperIdentity:helper===mocked,actual:actual.value,builtinIdentity:fs===dynamic,tag:fs[Symbol.toStringTag]??null,keys:Object.keys(fs),descriptor:{get:typeof descriptor.get,writable:descriptor.writable??null,enumerable:descriptor.enumerable,configurable:descriptor.configurable}}));expect(mocked.value).toBe(73);expect(actual.value).toBe(19);expect(fs).toBe(dynamic)})
