import { expect, test, vi } from 'vitest'
const observe = (label, value) =>
  console.log('WS2_CJS_MOCK_RAW:' + JSON.stringify({ label, ...value }))
test('automatic CommonJS object mocks preserve default and named export relationships', async () => {
  const first = await vi.importMock('./object.cjs')
  const second = await vi.importMock('././object.cjs')
  first.fn('first')
  first.default.fn('default')
  const original = await vi.importActual('./object.cjs')
  expect(first.value).toBe(42)
  expect(first.default.value).toBe(first.value)
  expect(vi.isMockFunction(first.fn)).toBe(true)
  expect(first.default.fn).toBe(first.fn)
  expect(first.fn.mock.calls).toEqual([['first'], ['default']])
  expect(second).not.toBe(first)
  expect(second.fn).not.toBe(first.fn)
  expect(second.fn.mock.calls).toEqual([])
  expect(vi.isMockFunction(original.fn)).toBe(false)
  expect(original.fn(2)).toBe(3)
  expect(globalThis.__ws2CjsObjectEvaluations).toBe(1)
  observe('object', { value: first.value, calls: first.fn.mock.calls })
})
test('automatic CommonJS esModule mocks preserve default function and named values', async () => {
  const first = await vi.importMock('./esmodule.cjs')
  const second = await vi.importMock('./esmodule.cjs')
  first.default(3)
  expect(vi.isMockFunction(first.default)).toBe(true)
  expect(first.named).toBe(17)
  expect(first.default(4)).toBeUndefined()
  expect(first.default.mock.calls).toEqual([[3], [4]])
  expect(second).not.toBe(first)
  expect(second.default).not.toBe(first.default)
  expect(second.default.mock.calls).toEqual([])
  observe('esmodule', { named: first.named, calls: first.default.mock.calls })
})
test('automatic CommonJS primitive mocks preserve the primitive default', async () => {
  const first = await vi.importMock('./primitive.cjs')
  const second = await vi.importMock('./primitive.cjs')
  expect(first.default).toBe(19)
  expect(second).not.toBe(first)
  observe('primitive', { value: first.default })
})
test('automatic CommonJS mocks preserve an authored falsy esModule marker', async () => {
  const mock = await vi.importMock('./falsy.cjs')
  expect(mock.__esModule).toBe(false)
  expect(mock.default.__esModule).toBe(false)
  expect(mock.value).toBe(29)
  expect(mock.default.value).toBe(mock.value)
  observe('falsy', { marker: mock.__esModule, value: mock.value })
})
