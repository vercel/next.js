function counts(form) {
  return {
    root: 0,
    dependency: 0,
    defaults: 0,
    keys: 0,
    ...globalThis.__cjsGrammarCounts?.[form.id],
  }
}

function describe(value) {
  return value === undefined
    ? { type: 'undefined' }
    : { type: typeof value, value }
}

function expectedBindings(form, scenario, args, evaluation) {
  const family = form.id.split(':')[0]
  const defaultValue = `dependency-${form.id}-1@${evaluation}`
  const usesDefault = parameters[family].defaults.includes(scenario.name)
  if (family === 'object-rest' || family === 'array-rest') {
    return {
      head: describe(usesDefault ? defaultValue : 0),
      value: describe(
        scenario.name === 'present'
          ? family === 'object-rest'
            ? { tail: 2, other: 3 }
            : [2, 3]
          : family === 'object-rest'
            ? {}
            : []
      ),
    }
  }
  let value
  if (family === 'destructured-rest-object') value = args.length
  else if (usesDefault) value = defaultValue
  else if (scenario.name === 'null-value' || scenario.name === 'null')
    value = null
  else if (scenario.name === 'empty') value = undefined
  else value = 0
  return { value: describe(value) }
}

function checkCallable(namespace, form, held = false) {
  const [family, kind, syntax] = form.id.split(':')
  const raw = namespace.default
  expect(typeof raw).toBe('function')
  expect(raw.name).toBe(
    held
      ? 'owned-renamed-callable'
      : syntax === 'named' || syntax === 'binding'
        ? 'callable'
        : ''
  )
  expect(raw.length).toBe(held ? 7 : parameters[family].length)
  const hasPrototype =
    kind === 'generator' ||
    kind === 'async-generator' ||
    (kind === 'sync' && syntax !== 'arrow')
  expect(Object.hasOwn(raw, 'prototype')).toBe(hasPrototype)
  if (kind === 'sync' && syntax !== 'arrow')
    expect(raw.prototype.constructor).toBe(raw)
  return raw
}

async function checkInvocations(namespace, form, evaluation, held = false) {
  const [family, kind, syntax] = form.id.split(':')
  const raw = checkCallable(namespace, form, held)
  for (const scenario of scenarios[family]) {
    const args = scenario.args()
    const before = counts(form)
    const execute = async () => {
      const result = Reflect.apply(
        raw,
        { receiver: 'authored', current: raw },
        args
      )
      if (kind === 'generator' || kind === 'async-generator') {
        const iterator =
          kind === 'generator' ? Symbol.iterator : Symbol.asyncIterator
        expect(result[iterator]()).toBe(result)
        const first = await result.next()
        expect(first.done).toBe(false)
        expect(await result.next()).toEqual({
          value: { evaluation, returned: true },
          done: true,
        })
        return first.value
      }
      expect(types.isPromise(result)).toBe(kind === 'async')
      return await result
    }
    if (parameters[family].errors.includes(scenario.name)) {
      await expect(execute()).rejects.toThrow(TypeError)
    } else {
      expect(await execute()).toEqual({
        evaluation,
        dependency: `dependency-${form.id}-1`,
        bindings: expectedBindings(form, scenario, args, evaluation),
        receiver: syntax === 'arrow' ? null : 'authored',
        argumentCount: syntax === 'arrow' ? null : args.length,
      })
    }
    const after = counts(form)
    expect(after.root).toBe(before.root)
    expect(after.dependency).toBe(before.dependency)
    expect(after.defaults - before.defaults).toBe(
      parameters[family].defaults.includes(scenario.name) ? 1 : 0
    )
    expect(after.keys - before.keys).toBe(
      family === 'computed-object' &&
        !parameters[family].errors.includes(scenario.name)
        ? 1
        : 0
    )
  }
}

for (const form of forms) {
  test(form.id + ':reset', async () => {
    const before = counts(form)
    vi.resetModules()
    const first = await form.load()
    const cached = await form.load()
    expect(cached).toBe(first)
    expect(cached.default).toBe(first.default)
    vi.resetModules()
    const fresh = await form.load()
    expect(fresh).not.toBe(first)
    expect(fresh.default).not.toBe(first.default)
    expect(counts(form)).toMatchObject({ root: before.root + 2, dependency: 1 })
    await checkInvocations(first, form, before.root + 1)
    await checkInvocations(cached, form, before.root + 1)
    await checkInvocations(fresh, form, before.root + 2)
  })

  test(form.id + ':actual', async () => {
    const before = counts(form)
    vi.resetModules()
    const first = await form.actual()
    const ordinary = await form.load()
    const again = await form.actual()
    expect(ordinary).toBe(first)
    expect(again).toBe(first)
    expect(again.default).toBe(first.default)
    vi.resetModules()
    const fresh = await form.actual()
    expect(fresh).not.toBe(first)
    expect(fresh.default).not.toBe(first.default)
    expect(counts(form)).toMatchObject({ root: before.root + 2, dependency: 1 })
    await checkInvocations(first, form, before.root + 1)
    await checkInvocations(ordinary, form, before.root + 1)
    await checkInvocations(again, form, before.root + 1)
    await checkInvocations(fresh, form, before.root + 2)
  })

  test(form.id + ':held', async () => {
    const before = counts(form)
    vi.resetModules()
    const first = await form.load()
    const raw = checkCallable(first, form)
    const extra = { held: true }
    raw.extra = extra
    Object.defineProperty(raw, 'name', {
      value: 'owned-renamed-callable',
      configurable: true,
    })
    Object.defineProperty(raw, 'length', { value: 7, configurable: true })
    if (raw.prototype !== undefined) raw.prototype.ownedPrototype = 'held'
    await checkInvocations(first, form, before.root + 1, true)
    vi.resetModules()
    const fresh = await form.load()
    expect(first.default).toBe(raw)
    expect(raw.extra).toBe(extra)
    expect(fresh.default).not.toBe(raw)
    expect(Object.hasOwn(fresh.default, 'extra')).toBe(false)
    if (raw.prototype !== undefined) {
      expect(raw.prototype.ownedPrototype).toBe('held')
      expect(Object.hasOwn(fresh.default.prototype, 'ownedPrototype')).toBe(
        false
      )
    }
    expect(counts(form)).toMatchObject({ root: before.root + 2, dependency: 1 })
    await checkInvocations(first, form, before.root + 1, true)
    await checkInvocations(fresh, form, before.root + 2)
  })
}
