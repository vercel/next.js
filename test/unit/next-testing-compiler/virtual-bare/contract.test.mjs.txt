import { expect, test, vi } from 'vitest'
vi.mock('ws2-completely-absent-package', () => ({
  value: 73,
  default: { source: 'virtual' },
}))
vi.mock('ws2-absent-async-package', async () => ({
  value: await Promise.resolve(91),
}))
vi.mock('@ws2/absent/subpath', () => ({ value: 117 }))
import fallback, { value } from 'ws2-completely-absent-package'
import * as named from 'ws2-completely-absent-package'
import {
  value as throughConsumer,
  fallback as consumerDefault,
} from './consumer.mjs'
import { value as asyncValue } from 'ws2-absent-async-package'
import { value as scopedValue } from '@ws2/absent/subpath'
function row(value) {
  console.log('WS2_VIRTUAL_RAW:' + JSON.stringify(value))
}

test('supplies a missing bare module through one actual factory across graph consumers', async () => {
  const dynamic = await import('ws2-completely-absent-package')
  expect(value).toBe(73)
  expect(throughConsumer).toBe(73)
  expect(fallback.source).toBe('virtual')
  expect(fallback).toBe(consumerDefault)
  expect(named).toBe(dynamic)
  row({ name: 'bare', value })
})
test('awaits an explicit missing bare module factory', () => {
  expect(asyncValue).toBe(91)
  row({ name: 'async', value: asyncValue })
})
test('supplies a scoped missing bare subpath factory', () => {
  expect(scopedValue).toBe(117)
  row({ name: 'scoped', value: scopedValue })
})
