import { chai, expect, test, afterAll } from 'vitest'
import assert from 'node:assert/strict'
function emit(row) {
  if ('sameCallable' in row) assert.equal(row.sameCallable, true)
  for (const call of row.calls) {
    assert.equal(call.success, true, call.error?.message)
    assert.equal(call.sameResult, true)
  }
  for (const value of row.observations) {
    const contextual = value.label !== 'raw'
    assert.equal(value.sameFirstTask, contextual)
    assert.equal(value.taskType, contextual ? 'test' : null)
    assert.equal(value.taskName, contextual ? firstTask.name : null)
  }
}
const observations = []
let firstTask, retained, assertions
chai.use((_chai, utils) =>
  utils.addMethod(
    _chai.Assertion.prototype,
    'toOwnerIdentity',
    function (expected) {
      const task = utils.flag(this, 'vitest-test')
      observations.push({
        label: this._obj.label,
        taskName: task?.name ?? null,
        taskType: task?.type ?? null,
        sameFirstTask: task === firstTask,
      })
      this.assert(
        this._obj.label === expected,
        'wrong label',
        'unexpected label',
        expected,
        this._obj.label
      )
      return this
    }
  )
)
function call(name, method, assertion, expected) {
  try {
    const result = method.call(assertion, expected)
    return { name, success: true, sameResult: result === assertion }
  } catch (error) {
    return {
      name,
      success: false,
      error: { name: error.name, message: error.message },
    }
  }
}
test('actual raw global and context assertion receiver origins', ({
  expect: local,
  task,
}) => {
  firstTask = task
  retained = chai.Assertion.prototype.toOwnerIdentity
  assertions = {
    raw: chai.expect({ label: 'raw' }),
    global: expect({ label: 'global' }),
    local: local({ label: 'local' }),
  }
  const calls = Object.entries(assertions).map(([name, assertion]) =>
    call(name, retained, assertion, name)
  )
  emit({ kind: 'first', calls, observations: observations.splice(0) })
})
test('retained assertion origins stay actual across a later task', () => {
  const current = chai.Assertion.prototype.toOwnerIdentity
  const calls = Object.entries(assertions).flatMap(([name, assertion]) => [
    call('retained-' + name, retained, assertion, name),
    call('fresh-' + name, current, assertion, name),
  ])
  emit({
    kind: 'next',
    sameCallable: current === retained,
    calls,
    observations: observations.splice(0),
  })
})
afterAll(() => {
  const calls = Object.entries(assertions).map(([name, assertion]) =>
    call('afterAll-' + name, retained, assertion, name)
  )
  emit({ kind: 'afterAll', calls, observations: observations.splice(0) })
})
