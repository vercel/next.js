import { chai, test } from 'vitest'
const trace = []
const emit = row => console.log('POLL_OWNERSHIP_ROW ' + JSON.stringify(row))
chai.use((c, utils) => {
  utils.overwriteMethod.bind(utils, c.Assertion.prototype, 'equal')((original) => function (value) {
    trace.push({ phase: 'before-super', subject: this._obj })
    const result = original.call(this, value)
    trace.push({ phase: 'after-super', subject: result._obj, sameResult: result === this })
    return result
  })
})
let retained
function action(method, receiver, value) {
  try { const result = method.call(receiver, value); return { passed: true, subject: result._obj, sameResult: result === receiver } }
  catch(error) { return { passed: false, error: { name: error.name, message: error.message, actual: error.actual, expected: error.expected } } }
}
test('same-task prototype overwrite delegates to native super with actual receiver', () => {
  retained = chai.Assertion.prototype.equal
  const value = chai.expect(1)
  emit({ kind: 'same-task', result: action(retained, value, 1), trace: trace.splice(0) })
})
test('retained prototype overwrite delegates with next actual receiver', () => {
  emit({ kind: 'later-task', sameCallable: retained === chai.Assertion.prototype.equal, result: action(retained, chai.expect(2), 2), trace: trace.splice(0) })
})
