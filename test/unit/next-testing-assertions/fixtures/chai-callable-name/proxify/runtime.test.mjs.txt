import { chai, expect, test, afterAll } from 'vitest'
import assert from 'node:assert/strict'
const marker = {}
let base,
  alias,
  reads = 0
chai.use((c, u) =>
  u.addMethod(c.Assertion.prototype, 'nameAliased', function () {
    return this
  })
)
function checkDeleted() {
  for (const fn of [base, alias]) {
    assert.equal(Object.hasOwn(fn, 'name'), false)
    assert.equal(fn.name, '')
  }
  assert.equal(reads, 2)
}
test('native proxify shares actual function metadata and getter receivers', () => {
  base = chai.Assertion.prototype.nameAliased
  Object.defineProperty(base, 'sharedNameField', {
    configurable: true,
    value: marker,
  })
  const getter = function () {
    reads++
    return this
  }
  Object.defineProperty(base, 'name', { configurable: true, get: getter })
  alias = chai.util.proxify(base)
  const first = base.name,
    second = alias.name
  assert.equal(alias === base, chai.config.useProxy === false)
  assert.equal(first, second)
  assert.equal(first === base, chai.config.useProxy === false)
  assert.equal(second === alias, chai.config.useProxy === false)
  assert.equal(first.sharedNameField, marker)
  assert.equal(second.sharedNameField, marker)
  assert.equal(Object.getOwnPropertyDescriptor(alias, 'name').get, getter)
  assert.equal(reads, 2)
  Object.defineProperty(alias, 'name', {
    configurable: true,
    value: 'actual shared renamed callable',
  })
  assert.equal(base.name, 'actual shared renamed callable')
  assert.equal(alias.name, base.name)
  assert.equal(Object.getOwnPropertyDescriptor(base, 'name').value, base.name)
})
test('retained aliases preserve real deletion and shared inherited name', () => {
  assert.equal(delete base.name, true)
  checkDeleted()
})
afterAll(checkDeleted)
