import { test, expect, beforeEach, afterEach, afterAll } from 'vitest'
const rows = []
let release = () => {}
const row = (stage, value = {}) => rows.push({ stage, ...value })
afterEach(({ task }) => { row('afterEach', { errors: (task.result?.errors ?? []).map(error => error?.message ?? String(error)) }); release() })
afterAll(() => console.log('TASK_QUEUE_CONSUMER:' + JSON.stringify(rows)))
beforeEach(({ task, expect: local }) => {
  let yes
  release = () => { row('release-native'); yes(17) }
  local(new Promise(resolve => { yes = resolve })).resolves.toBe(17).then(() => row('native-settled'))
  row('beforeEach-return', { length: task.promises.length })
})
test('actual public body queue also owns beforeEach native promises', ({ task }) => { task.promises = []; row('body-return') })
