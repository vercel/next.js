import { test, expect, afterEach, afterAll } from 'vitest'
const rows = []
let release = () => {}
const row = (stage, value = {}) => rows.push({ stage, ...value })
afterEach(({ task }) => { row('afterEach', { errors: (task.result?.errors ?? []).map(error => error?.message ?? String(error)) }); release() })
afterAll(() => console.log('TASK_QUEUE_CONSUMER:' + JSON.stringify(rows)))
test('actual public manual thenable releases native assertion', ({ task, expect: local }) => {
  let yes, released = false
  const native = new Promise(resolve => { yes = resolve })
  release = () => { if (!released) { released = true; row('release-native'); yes(17) } }
  const pending = local(native).resolves.toBe(17)
  pending.then(() => row('native-settled'))
  task.promises.push({ then(resolve) { row('manual-then'); release(); resolve(29) } })
  row('body-return', { length: task.promises.length })
})
