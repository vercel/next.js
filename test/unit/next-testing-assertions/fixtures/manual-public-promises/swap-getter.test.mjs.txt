import { test, expect, afterEach, afterAll } from 'vitest'
const rows = []
let release = () => {}
const row = (stage, value = {}) => rows.push({ stage, ...value })
afterEach(({ task }) => { row('afterEach', { errors: (task.result?.errors ?? []).map(error => error?.message ?? String(error)) }); release() })
afterAll(() => console.log('TASK_QUEUE_CONSUMER:' + JSON.stringify(rows)))
test('actual public getter selects second iterable', ({ task }) => {
  const first = [{ then(resolve) { row('first-then'); resolve(1) } }]
  const second = [{ then(resolve) { row('second-then'); resolve(2) } }]
  let reads = 0
  Object.defineProperty(task, 'promises', { configurable: true, get() { row('queue-get', { read: ++reads }); return reads === 1 ? first : second } })
  row('body-return')
})
