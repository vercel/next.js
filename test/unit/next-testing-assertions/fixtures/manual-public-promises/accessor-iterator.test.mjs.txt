import { test, expect, afterEach, afterAll } from 'vitest'
const rows = []
let release = () => {}
const row = (stage, value = {}) => rows.push({ stage, ...value })
afterEach(({ task }) => { row('afterEach', { errors: (task.result?.errors ?? []).map(error => error?.message ?? String(error)) }); release() })
afterAll(() => console.log('TASK_QUEUE_CONSUMER:' + JSON.stringify(rows)))
test('actual public queue accessor and iterator', ({ task }) => {
  const thenable = { get then() { row('then-get'); return function(resolve) { row('then-call', { sameThis: this === thenable, args: arguments.length }); resolve(17) } } }
  const queue = [thenable]
  const iterator = queue[Symbol.iterator]
  Object.defineProperty(queue, Symbol.iterator, { get() { row('iterator-get'); return function() { row('iterator-call', { sameThis: this === queue }); return iterator.call(this) } } })
  let reads = 0
  Object.defineProperty(task, 'promises', { configurable: true, get() { row('queue-get', { read: ++reads }); return queue } })
  row('body-return')
})
