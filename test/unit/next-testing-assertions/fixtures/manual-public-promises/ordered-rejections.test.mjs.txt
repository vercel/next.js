import { test, expect, afterEach, afterAll } from 'vitest'
const rows = []
let release = () => {}
const row = (stage, value = {}) => rows.push({ stage, ...value })
afterEach(({ task }) => { row('afterEach', { errors: (task.result?.errors ?? []).map(error => error?.message ?? String(error)) }); release() })
afterAll(() => console.log('TASK_QUEUE_CONSUMER:' + JSON.stringify(rows)))
test('actual public queue rejection order', ({ task }) => {
  const first = new Error('queue first'); first.stack = ''
  const second = new Error('queue second'); second.stack = ''
  task.promises = [Promise.reject(first), Promise.reject(0), Promise.reject(second)]
  row('body-return')
})
