import { test, expect, afterEach, afterAll } from 'vitest'
const rows = []
let release = () => {}
const row = (stage, value = {}) => rows.push({ stage, ...value })
afterEach(({ task }) => { row('afterEach', { errors: (task.result?.errors ?? []).map(error => error?.message ?? String(error)) }); release() })
afterAll(() => console.log('TASK_QUEUE_CONSUMER:' + JSON.stringify(rows)))
test('actual public empty replacement releases body before native gate', ({ task, expect: local }) => {
  let yes
  const native = new Promise(resolve => { yes = resolve })
  release = () => { row('release-native'); yes(17) }
  local(native).resolves.toBe(17).then(() => row('native-settled'))
  task.promises = []
  row('body-return')
})
