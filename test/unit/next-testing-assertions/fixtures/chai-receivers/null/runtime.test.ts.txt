import { test, chai, expect } from 'vitest'
import assert from 'node:assert/strict'
const proxify = (value) =>
  Reflect.apply(chai.util.proxify, chai.util, [value, 'equal'])
for (const useProxy of [true, false])
  for (const kind of [
    'method',
    'proxied-method',
    'prototype',
    'proxied-prototype',
  ])
    for (const receiver of [undefined, null])
      test(
        (useProxy ? 'enabled ' : 'disabled ') + kind + ' ' + String(receiver),
        () => {
          const previous = chai.config.useProxy
          chai.config.useProxy = useProxy
          try {
            const assertion = chai.expect(3)
            let method = kind.includes('prototype')
              ? Reflect.get(chai.Assertion.prototype, 'equal')
              : assertion.equal
            if (kind.startsWith('proxied')) method = proxify(method)
            assert.throws(
              () => Reflect.apply(method, receiver, [3]),
              (error) =>
                error?.name === 'TypeError' &&
                error.message.includes(String(receiver))
            )
            assert.equal(expect.getState().assertionCalls, 0)
          } finally {
            chai.config.useProxy = previous
          }
        }
      )
