import { test, chai, expect } from 'vitest'
import assert from 'node:assert/strict'
const classes = [
  ['raw', (v) => chai.expect(v), 'equal'],
  ['expect-equal', (v) => expect(v), 'equal'],
  ['expect-toBe', (v) => expect(v), 'toBe'],
  ['constructor', (v) => new chai.Assertion(v), 'equal'],
]
for (const useProxy of [true, false])
  for (const [kind, make, key] of classes)
    for (const operation of ['detached', 'self', 'other'])
      test(
        (useProxy ? 'enabled ' : 'disabled ') + kind + ' ' + operation,
        () => {
          const previous = chai.config.useProxy
          chai.config.useProxy = useProxy
          try {
            const assertion = make(1),
              other = make(3),
              method = assertion[key]
            const receiver =
              operation === 'self'
                ? assertion
                : operation === 'other'
                  ? other
                  : undefined
            const value = operation === 'self' ? 1 : 3
            if (operation === 'detached')
              assert.throws(
                () => Reflect.apply(method, receiver, [value]),
                (error) =>
                  error?.name === 'TypeError' && /undefined/.test(error.message)
              )
            else {
              const result = Reflect.apply(method, receiver, [value])
              assert.notEqual(result, assertion)
              assert.notEqual(result, other)
              assert.equal(typeof result, 'object')
            }
            assert.equal(
              expect.getState().assertionCalls,
              kind.startsWith('expect-') ? 2 : 0
            )
          } finally {
            chai.config.useProxy = previous
          }
        }
      )
