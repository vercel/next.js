import { chai, expect, test, afterAll } from 'vitest'
import assert from 'node:assert/strict'
chai.config.useProxy = false
const marker = {},
  callerError = new Error('actual caller data function error'),
  foreign = {},
  retained = {},
  aliasRetained = {},
  trace = []
const flags = [
  ['fixed', false, false],
  ['configurable', false, true],
  ['writable', true, false],
  ['mutable', true, true],
]
let host,
  directAlias,
  originalInput,
  originalExpect,
  neverReads = 0,
  payloadReads = 0,
  replacement,
  boundAlias
const neverGetter = function () {
  neverReads++
  throw new Error('never-called metadata getter')
}
function payload(value) {
  trace.push({ receiver: this, value })
  if (value === 'throw') throw callerError
  return marker
}
Object.defineProperty(payload, 'neverMetadata', {
  configurable: true,
  get() {
    payloadReads++
    throw new Error('never-called payload metadata getter')
  },
})
chai.use((c, u) => {
  u.addMethod(c.Assertion.prototype, 'dataHost', function () {
    return this
  })
  u.addMethod(c.Assertion.prototype, 'dataAlias', function () {
    trace.push({ subject: u.flag(this, 'object') })
    return this
  })
})
function callPayload(fn, receiver) {
  assert.equal(Reflect.apply(fn, receiver, ['value']), marker)
  assert.equal(trace.at(-1).receiver, receiver)
  assert.throws(
    () => Reflect.apply(fn, receiver, ['throw']),
    (error) => error === callerError
  )
  assert.equal(trace.at(-1).receiver, receiver)
}
function callAlias(fn, receiver, subject) {
  assert.equal(Reflect.apply(fn, receiver, ['value']), receiver)
  assert.equal(trace.at(-1).subject, subject)
}
test('own function data preserves descriptor identity, receivers and original errors', ({
  expect: taskExpect,
}) => {
  originalExpect = taskExpect
  originalInput = taskExpect(17)
  host = chai.Assertion.prototype.dataHost
  directAlias = chai.Assertion.prototype.dataAlias
  Object.defineProperty(host, 'neverField', {
    get: neverGetter,
    configurable: true,
    enumerable: true,
  })
  for (const [key, writable, configurable] of flags) {
    Object.defineProperty(host, key, {
      value: payload,
      writable,
      configurable,
      enumerable: true,
    })
    Object.defineProperty(host, key + 'Alias', {
      value: directAlias,
      writable,
      configurable,
      enumerable: true,
    })
    assert.deepEqual(Object.getOwnPropertyDescriptor(host, key), {
      value: payload,
      writable,
      configurable,
      enumerable: true,
    })
    assert.equal(host[key], payload)
    assert.equal(host[key + 'Alias'], directAlias)
    retained[key] = host[key]
    aliasRetained[key] = host[key + 'Alias']
    assert.equal(host[key]('method'), marker)
    assert.equal(trace.at(-1).receiver, host)
    callPayload(retained[key], foreign)
    callPayload(retained[key], undefined)
    callAlias(aliasRetained[key], originalInput, 17)
  }
  assert.equal(originalExpect.getState().assertionCalls, 1)
})
test('retained data callbacks and public aliases preserve current and original receivers', ({
  expect: taskExpect,
}) => {
  const currentInput = taskExpect(33)
  for (const [key] of flags) {
    assert.equal(retained[key], payload)
    callPayload(retained[key], foreign)
    assert.equal(aliasRetained[key], directAlias)
    callAlias(aliasRetained[key], originalInput, 17)
    callAlias(aliasRetained[key], currentInput, 33)
  }
  assert.equal(originalExpect.getState().assertionCalls, 1)
  assert.equal(taskExpect.getState().assertionCalls, 1)
})
test('data mutations retain old callbacks and preserve bound public aliases and symbols', ({
  expect: taskExpect,
}) => {
  const alternate = {},
    replacementError = new Error('actual replacement function error')
  replacement = function (value) {
    trace.push({ receiver: this, value })
    if (value === 'throw') throw replacementError
    return alternate
  }
  assert.equal(Reflect.set(host, 'fixed', replacement), false)
  Object.defineProperty(host, 'configurable', { value: replacement })
  assert.equal(Reflect.set(host, 'writable', replacement), true)
  assert.equal(Reflect.set(host, 'mutable', replacement), true)
  assert.equal(host.fixed, payload)
  for (const key of ['configurable', 'writable', 'mutable']) {
    assert.equal(host[key], replacement)
    assert.equal(Reflect.apply(host[key], foreign, ['value']), alternate)
    assert.equal(trace.at(-1).receiver, foreign)
    assert.throws(
      () => Reflect.apply(host[key], foreign, ['throw']),
      (error) => error === replacementError
    )
  }
  assert.throws(() => directAlias.bind(originalInput), /dataAlias.length/)
  boundAlias = function (...args) {
    return Reflect.apply(directAlias, originalInput, args)
  }
  assert.equal(Reflect.set(host, 'fixedAlias', boundAlias), false)
  Object.defineProperty(host, 'configurableAlias', { value: boundAlias })
  Reflect.set(host, 'writableAlias', boundAlias)
  Reflect.set(host, 'mutableAlias', boundAlias)
  const currentInput = taskExpect(71)
  for (const key of ['configurableAlias', 'writableAlias', 'mutableAlias']) {
    assert.equal(host[key], boundAlias)
    assert.equal(
      Reflect.apply(host[key], currentInput, ['bound']),
      originalInput
    )
    assert.equal(trace.at(-1).subject, 17)
  }
  const symbol = Symbol.for('next.caller.function.data')
  Object.defineProperty(host, symbol, {
    value: replacement,
    writable: false,
    configurable: false,
  })
  assert.equal(host[symbol], replacement)
  const revoked = Proxy.revocable(function () {}, {})
  revoked.revoke()
  Object.defineProperty(host, 'revokedFunction', {
    value: revoked.proxy,
    writable: false,
    configurable: false,
  })
  assert.equal(host.revokedFunction, revoked.proxy)
  assert.throws(
    () => Reflect.apply(host.revokedFunction, undefined, []),
    (error) => error instanceof TypeError && /revoked/.test(error.message)
  )
  assert(Object.values(retained).every((value) => value === payload))
})
afterAll(() => {
  assert.equal(
    Object.getOwnPropertyDescriptor(host, 'neverField').get,
    neverGetter
  )
  for (const [key] of flags) {
    assert.equal(
      Object.getOwnPropertyDescriptor(host, key).value,
      key === 'fixed' || !replacement ? payload : replacement
    )
    assert.equal(
      Object.getOwnPropertyDescriptor(host, key + 'Alias').value,
      key === 'fixed' || !boundAlias ? directAlias : boundAlias
    )
  }
  assert.equal(originalExpect.getState().assertionCalls, 1)
  assert.equal(neverReads, 0)
  assert.equal(payloadReads, 0)
})
