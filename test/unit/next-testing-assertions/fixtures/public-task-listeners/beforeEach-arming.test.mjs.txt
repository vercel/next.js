import { test, expect, beforeEach, afterEach, afterAll, onTestFinished as globalFinished } from 'vitest'
const rows = []
let release = () => {}
const row = (stage, value = {}) => rows.push({ stage, ...value })
afterEach(({ task }) => { row('afterEach', { errors: (task.result?.errors ?? []).map(error => error?.message ?? String(error)) }); release() })
afterAll(() => console.log('TASK_QUEUE_CONSUMER:' + JSON.stringify(rows)))
beforeEach(({task, expect: local}) => {
   local(Promise.resolve(17)).resolves.toBe(17).then(() => row('native-settled'))
   row('beforeEach-native', { array: Array.isArray(task.onFinished), length: task.onFinished?.length, ownPush: task.onFinished && Object.hasOwn(task.onFinished, 'push') })
 })
 test('actual native finished observer arms before beforeEach', ({task}) => row('body-return', { length: task.onFinished.length }))
