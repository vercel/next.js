vi.mock('./target.cjs',()=>{globalThis.__requireReset126FactoryCount=(globalThis.__requireReset126FactoryCount??0)+1;return{value:'mocked',default:{value:'mock-default',token:{}}}})
const reset = vi.resetModules

test('reset composes the original require cache with the mocked ESM namespace',async()=>{
 const beforeRaw=required.read();
 const beforeMocked=await import('./target.cjs');
 const beforeConsumer=await import('./consumer.cjs');
 const returned=vi.resetModules();
 const afterMocked=await import('./target.cjs');
 const afterConsumer=await import('./consumer.cjs');
 const heldRaw=required.read();
 const afterRaw=afterConsumer.default.read();
 expect(heldRaw.value).toBe('actual');expect(afterRaw.value).toBe('actual');expect(afterMocked.value).toBe('mocked');
 appendFileSync(process.env.NEXT_REQUIRE_ROWS,JSON.stringify({name:'reset-require-composition',resetReturnSame:returned===vi,rawHeldIdentity:heldRaw===beforeRaw,rawReloadedIdentity:afterRaw===beforeRaw,rawTokenIdentity:afterRaw.token===beforeRaw.token,consumerDefaultIdentity:afterConsumer.default===required,consumerNamespaceIdentity:afterConsumer===beforeConsumer,consumerInitialIdentity:afterConsumer.default.initial===beforeRaw,mockedNamespaceIdentity:afterMocked===beforeMocked,staticNamespaceIdentity:afterMocked===mocked,defaultTokenIdentity:afterMocked.default.token===beforeMocked.default.token,factoryCalls:globalThis.__requireReset126FactoryCount,rawValue:afterRaw.value,mockedValue:afterMocked.value,defaultValue:afterMocked.default.value,beforeRawDescriptors:descriptor(beforeRaw),afterRawDescriptors:descriptor(afterRaw),beforeMockedDescriptors:descriptor(beforeMocked),afterMockedDescriptors:descriptor(afterMocked),beforeConsumerDescriptors:descriptor(beforeConsumer),afterConsumerDescriptors:descriptor(afterConsumer),beforeRequiredDescriptors:descriptor(required),afterRequiredDescriptors:descriptor(afterConsumer.default)})+'\n')
})

import requiredB from './consumer-b.cjs'
import requiredC from './consumer-c.cjs'
const same=(values)=>values.map(value=>values.map(other=>value===other))
test('multiple CommonJS consumers share one original leaf across reset',async()=>{
 const beforeNamespaces=[await import('./consumer.cjs'),await import('./consumer-b.cjs'),await import('./consumer-c.cjs')]
 const held=[required,requiredB,requiredC]
 const before=beforeNamespaces.map(value=>value.default)
 const beforeRaw=[...held,...before].map(value=>value.read())
 const returned=reset()
 const afterNamespaces=[await import('./consumer.cjs'),await import('./consumer-b.cjs'),await import('./consumer-c.cjs')]
 const after=afterNamespaces.map(value=>value.default)
 const afterRaw=[...held,...after].map(value=>value.read())
 const raw=[...beforeRaw,...afterRaw]
 expect(raw.map(value=>value.value)).toEqual(Array(raw.length).fill('actual'))
 expect(mocked.value).toBe('mocked')
 appendFileSync(process.env.NEXT_REQUIRE_ROWS,JSON.stringify({name:'multiple-consumer-graph',resetReturnSame:returned===vi,consumerIdentities:same([...held,...before,...after]),namespaceIdentities:same([...beforeNamespaces,...afterNamespaces]),rawIdentities:same(raw),rawTokenIdentities:same(raw.map(value=>value.token)),rawValues:raw.map(value=>value.value),factoryCalls:globalThis.__requireReset126FactoryCount,consumers:[...held,...before,...after].map(value=>({keys:Reflect.ownKeys(value).map(String),descriptors:descriptor(value),initialIsFirstRaw:value.initial===raw[0]})),mockDescriptors:descriptor(mocked)})+'\n')
})
