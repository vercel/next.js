import { test, expect, vi } from 'vitest'
import * as mocked from './target.cjs'
import required from './consumer.cjs'
vi.mock('./target.cjs', async function () {
  globalThis.__requireReset126FactoryCount =
    (globalThis.__requireReset126FactoryCount ?? 0) + 1
  return { value: 'mocked', default: { value: 'mock-default', token: {} } }
})
const reset = vi.resetModules
test('original CommonJS require and mocked ESM remain separate', async () => {
  const raw = required.read(),
    dynamic = await import('./target.cjs')
  expect(raw).toBe(required.initial)
  expect(raw.value).toBe('actual')
  expect(mocked.value).toBe('mocked')
  expect(mocked.default.value).toBe('mock-default')
  expect(dynamic).toBe(mocked)
  expect(globalThis.__requireReset126FactoryCount).toBe(1)
})

test('reset composes the original require cache with the mocked ESM namespace', async () => {
  const beforeRaw = required.read()
  const beforeMocked = await import('./target.cjs')
  const beforeConsumer = await import('./consumer.cjs')
  const returned = reset()
  const afterMocked = await import('./target.cjs')
  const afterConsumer = await import('./consumer.cjs')
  const heldRaw = required.read()
  const afterRaw = afterConsumer.default.read()
  expect(heldRaw.value).toBe('actual')
  expect(afterRaw.value).toBe('actual')
  expect(afterMocked.value).toBe('mocked')
  expect(returned).toBe(vi)
  expect(heldRaw).toBe(beforeRaw)
  expect(afterRaw).toBe(beforeRaw)
  expect(afterRaw.token).toBe(beforeRaw.token)
  expect(afterConsumer.default).not.toBe(required)
  expect(afterConsumer).not.toBe(beforeConsumer)
  expect(afterConsumer.default.initial).toBe(beforeRaw)
  expect(afterMocked).toBe(beforeMocked)
  expect(afterMocked).toBe(mocked)
  expect(afterMocked.default.token).toBe(beforeMocked.default.token)
  expect(globalThis.__requireReset126FactoryCount).toBe(1)
  expect(afterMocked.default.value).toBe('mock-default')
})
