const offset: number = 2;
function prefix(value: number): number { return value + 1; }
export class Counter {
  declare total: number;
  constructor(value: number) { this.total = value; }
  readonly read = function (delta: number): number { return this.total + delta; };
  static readonly combine = function (value: number): number { return value * 2; };
  method(delta: number): number { return this.total + delta; }
}
export function exercise(value: number): number {
  const counter = new Counter(prefix(value));
  return counter.read(offset) + Counter.combine(value) + counter.method(offset);
}
