import { expect, test } from 'vitest'
import { Card as Card0 } from './creator-classic-expression.tsx'
import { Card as Card1 } from './creator-classic-declaration.tsx'
import { Card as Card2 } from './creator-classic-classexpression.tsx'

test('classic expression preserves actual component construction and props',()=>{const yes=Card0({active:true,label:'hello'}),no=Card0({active:false,label:'world'});for(const [element,label]of [[yes.props.children,'hello'],[no.props.children,'fallback']]){const output=element.type(element.props);expect(output.type).toBe('b');expect(output.props.children).toBe(label)}})
test('classic declaration preserves actual component construction and props',()=>{const yes=Card1({active:true,label:'hello'}),no=Card1({active:false,label:'world'});for(const [element,label]of [[yes.props.children,'hello'],[no.props.children,'fallback']]){const output=new element.type(element.props).render();expect(output.type).toBe('b');expect(output.props.children).toBe(label)}})
