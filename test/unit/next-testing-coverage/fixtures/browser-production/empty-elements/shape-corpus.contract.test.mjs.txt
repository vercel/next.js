test('classic empty preserves authored child and prop behavior',()=>{const yes=Card0({active:true,label:'hello'}),no=Card0({active:false,label:'world'});expect(yes.type).toBe('input');expect(yes.props).toEqual({});expect(no.props).toEqual({})})
test('classic attributes preserves authored child and prop behavior',()=>{const yes=Card1({active:true,label:'hello'}),no=Card1({active:false,label:'world'});expect(yes.props).toEqual({title:'on',value:'hello'});expect(no.props).toEqual({title:'off',value:'world'})})

test('classic nested preserves authored child and prop behavior',()=>{const yes=Card4({active:true,label:'hello'}),no=Card4({active:false,label:'world'});expect(yes.props.children.type).toBe('input');expect(yes.props.children.props.title).toBe('hello');expect(no.props.children.type).toBe('br');expect(no.props.children.props).toEqual({})})
