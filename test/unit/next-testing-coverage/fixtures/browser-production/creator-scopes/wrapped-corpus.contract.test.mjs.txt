import { expect, test } from 'vitest'
import { View as View0 } from './wrapped-classic-expression.tsx'
import { View as View1 } from './wrapped-classic-declaration.tsx'
import { View as View2 } from './wrapped-classic-classexpression.tsx'
import { View as View3 } from './wrapped-automatic-expression.tsx'
import { View as View4 } from './wrapped-automatic-declaration.tsx'

test('classic expression preserves real callable/class method execution',()=>{for(const label of ['hello','world']){const output=View0({label});expect(output.type).toBe('b');expect(output.props.children).toBe(label)}})
test('classic declaration preserves real callable/class method execution',()=>{for(const label of ['hello','world']){const output=new View1({label}).render();expect(output.type).toBe('b');expect(output.props.children).toBe(label)}})
