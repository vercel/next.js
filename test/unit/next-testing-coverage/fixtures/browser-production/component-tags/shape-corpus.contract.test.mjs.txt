test('classic single preserves authored child and prop behavior',()=>{const yes=Card0({active:true,label:'hello'}),no=Card0({active:false,label:'world'});expect(yes.type.name).toBe('Badge');expect(yes.type(yes.props).props.children).toBe('hello');expect(no.type(no.props).props.children).toBe('world')})

test('classic nested preserves authored child and prop behavior',()=>{const yes=Card3({active:true,label:'hello'}),no=Card3({active:false,label:'world'});expect(yes.props.children.map(node=>node.type(node.props).props.children)).toEqual(['hello','yes']);expect(no.props.children.map(node=>node.type(node.props).props.children)).toEqual(['world','no'])})
