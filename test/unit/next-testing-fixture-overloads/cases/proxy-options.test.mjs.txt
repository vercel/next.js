import { test as base, expect } from 'vitest'
const label = 'proxy-options'
const trace = []
const options = new Proxy(
  { auto: false, scope: 'test', injected: false },
  {
    ownKeys(target) {
      trace.push('keys')
      return Reflect.ownKeys(target)
    },
    getOwnPropertyDescriptor(target, key) {
      trace.push(['descriptor', String(key)])
      return Reflect.getOwnPropertyDescriptor(target, key)
    },
    get(target, key, receiver) {
      trace.push(['get', String(key)])
      return Reflect.get(target, key, receiver)
    },
    has(target, key) {
      trace.push(['has', String(key)])
      return Reflect.has(target, key)
    },
  }
)
const owned = base.extend({ value: [49, options] })
const registration = trace.slice()
owned(label, ({ value }) => {
  expect(value).toBe(49)
  expect(registration).toEqual([
    'keys',
    ['descriptor', 'auto'],
    ['descriptor', 'scope'],
    ['descriptor', 'injected'],
    ['get', 'auto'],
    ['get', 'scope'],
    ['get', 'injected'],
  ])
  expect(trace).toEqual(registration)
})
