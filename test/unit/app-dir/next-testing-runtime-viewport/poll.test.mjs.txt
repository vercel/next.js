import { test, expect } from 'vitest'

test('retains ordinary lazy poll facade shape and shorthand method data', async () => {
  let calls = 0
  const facade = expect
    .poll(() => {
      calls++
      return 7
    })
    .toBe(7)
  expect(calls).toBe(0)
  for (const method of ['then', 'catch', 'finally'])
    expect(typeof facade[method]).toBe('function')
  expect(facade[Symbol.toStringTag]).toBe('Promise')
  await facade
  expect(calls).toBe(1)
})

test('reads a replaced poll then getter once and invokes its actual receiver', async () => {
  let calls = 0,
    reads = 0,
    invocations = 0,
    getterReceiver = false,
    methodReceiver = false
  const facade = expect
      .poll(() => {
        calls++
        return 8
      })
      .toBe(8),
    original = facade.then
  Object.defineProperty(facade, 'then', {
    configurable: true,
    get() {
      reads++
      getterReceiver = this === facade
      return function (...args) {
        invocations++
        methodReceiver = this === facade
        return Reflect.apply(original, this, args)
      }
    },
  })
  await facade
  expect(reads).toBe(1)
  expect(invocations).toBe(1)
  expect(calls).toBe(1)
  expect(getterReceiver).toBe(true)
  expect(methodReceiver).toBe(true)
})

test('keeps a borrowed poll then method on the supplied receiver', async () => {
  let calls = 0
  const facade = expect
      .poll(() => {
        calls++
        return 9
      })
      .toBe(9),
    method = facade.then,
    receiver = { marker: 'borrowed' }
  await new Promise((resolve, reject) =>
    Reflect.apply(method, receiver, [resolve, reject])
  )
  expect(calls).toBe(1)
  expect(facade.then).toBe(method)
  expect(receiver.marker).toBe('borrowed')
})

test('keeps proxied poll values on the genuine getter and then invocation path', async () => {
  let calls = 0,
    reads = 0,
    invocations = 0,
    receiverIsProxy = false
  const facade = expect
      .poll(() => {
        calls++
        return 10
      })
      .toBe(10),
    original = facade.then
  const proxy = new Proxy(facade, {
    get(target, key, receiver) {
      if (key !== 'then') return Reflect.get(target, key, receiver)
      reads++
      return function (...args) {
        invocations++
        receiverIsProxy = this === proxy
        return Reflect.apply(original, this, args)
      }
    },
  })
  await proxy
  expect(calls).toBe(1)
  expect(reads).toBe(1)
  expect(invocations).toBe(1)
  expect(receiverIsProxy).toBe(true)
})
