import { test, expect, vi } from 'vitest'
import { utils } from 'vitest/browser'
const emit=value=>console.log('BROWSER_NATIVE_FIXTURE:'+JSON.stringify(value))

test('observes actual runtime environment changes at the default print limit boundary',()=>{
 const host=document.createElement('div');host.innerHTML='<span>debug environment boundary content</span>';document.body.append(host)
 try{
  const before=utils.prettyDOM(host,undefined,{highlight:false})
  vi.stubEnv('DEBUG_PRINT_LIMIT','8')
  const during=utils.prettyDOM(host,undefined,{highlight:false})
  vi.unstubAllEnvs()
  const after=utils.prettyDOM(host,undefined,{highlight:false})
  expect(typeof before).toBe('string');expect(typeof during).toBe('string');expect(after).toBe(before)
  emit({before,during,after,changed:during!==before})
 }finally{vi.unstubAllEnvs();host.remove()}
})

test('reads configured defaults at actual formatting calls and bypasses them for explicit zero',()=>{
 const host=document.createElement('div');host.textContent='configured getter boundary';document.body.append(host)
 const reads=[],defaults={highlight:false}
 Object.defineProperty(defaults,'maxLength',{enumerable:true,get(){reads.push('maxLength');return 10000}})
 try{
  utils.configurePrettyDOM(defaults)
  const zero=utils.prettyDOM(host,0),afterZero=[...reads]
  const defaultValue=utils.prettyDOM(host),afterDefault=[...reads]
  const explicitValue=utils.prettyDOM(host,10000),afterExplicit=[...reads]
  expect(zero).toBe('');expect(defaultValue).toBe(explicitValue)
  emit({zero,defaultValue,explicitValue,afterZero,afterDefault,afterExplicit})
 }finally{utils.configurePrettyDOM(undefined);host.remove()}
})
