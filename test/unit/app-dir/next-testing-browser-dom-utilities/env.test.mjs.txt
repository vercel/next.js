import { test, expect, vi } from 'vitest'
import { utils } from 'vitest/browser'
import expected from './dom-formatting.expected.json'

test('observes actual runtime environment changes at the default print limit boundary', () => {
  const host = document.createElement('div')
  host.innerHTML = '<span>debug environment boundary content</span>'
  document.body.append(host)
  try {
    const before = utils.prettyDOM(host, undefined, { highlight: false })
    vi.stubEnv('DEBUG_PRINT_LIMIT', '8')
    const during = utils.prettyDOM(host, undefined, { highlight: false })
    vi.unstubAllEnvs()
    const after = utils.prettyDOM(host, undefined, { highlight: false })
    expect(before).toBe(expected.environmentFull)
    expect(during).toBe(expected.environmentLimited)
    expect(after).toBe(before)
  } finally {
    vi.unstubAllEnvs()
    host.remove()
  }
})

test('reads configured defaults at actual formatting calls and bypasses them for explicit zero', () => {
  const host = document.createElement('div')
  host.textContent = 'configured getter boundary'
  document.body.append(host)
  const reads = [],
    defaults = { highlight: false }
  Object.defineProperty(defaults, 'maxLength', {
    enumerable: true,
    get() {
      reads.push('maxLength')
      return 10000
    },
  })
  try {
    utils.configurePrettyDOM(defaults)
    const zero = utils.prettyDOM(host, 0),
      afterZero = [...reads]
    const defaultValue = utils.prettyDOM(host),
      afterDefault = [...reads]
    const explicitValue = utils.prettyDOM(host, 10000),
      afterExplicit = [...reads]
    expect(zero).toBe('')
    expect(defaultValue).toBe(expected.getterDefaults)
    expect(explicitValue).toBe(expected.getterDefaults)
    expect(afterZero).toEqual([])
    expect(afterDefault).toEqual(['maxLength', 'maxLength'])
    expect(afterExplicit).toEqual(['maxLength', 'maxLength', 'maxLength'])
  } finally {
    utils.configurePrettyDOM(undefined)
    host.remove()
  }
})
