import { test, expect } from 'vitest'
import { page, utils } from 'vitest/browser'
import expected from './dom-formatting.expected.json'
const create = () => {
  const host = document.createElement('section')
  host.dataset.testid = 'dom-utility'
  host.innerHTML =
    '<h2>Hello <span>world</span></h2><button aria-label="Save">go</button>'
  document.body.append(host)
  return host
}

test('prints the actual owned DOM element and Locator with explicit options', () => {
  const host = create()
  try {
    const locator = page.getByTestId('dom-utility'),
      direct = utils.prettyDOM(host, 10000, { highlight: false }),
      located = utils.prettyDOM(locator, 10000, { highlight: false }),
      compact = utils.prettyDOM(host, 10000, { highlight: false, min: true })
    expect(direct).toBe(expected.full)
    expect(located).toBe(expected.full)
    expect(compact).toBe(expected.compact)
    expect(utils.prettyDOM(host, 0)).toBe('')
    expect(locator.element()).toBe(host)
  } finally {
    host.remove()
  }
})

test('retains actual configurable defaults and later authored object mutations', () => {
  const host = create(),
    defaults = { highlight: false, maxLength: 10000 }
  try {
    const returned = utils.configurePrettyDOM(defaults),
      full = utils.prettyDOM(host)
    defaults.maxLength = 24
    const clipped = utils.prettyDOM(host),
      explicit = utils.prettyDOM(host, 10000, { highlight: false, min: true })
    defaults.maxLength = 10000
    const recovered = utils.prettyDOM(host, 10000, {
      highlight: false,
      min: true,
    })
    expect(returned).toBeUndefined()
    expect(full).toBe(expected.full)
    expect(clipped).toBe(expected.clippedDefaults)
    expect(explicit).toBe(expected.compactWithDefaults)
    expect(recovered).toBe(expected.compact)
  } finally {
    if (typeof utils.configurePrettyDOM === 'function')
      utils.configurePrettyDOM(undefined)
    host.remove()
  }
})

test('debugs an actual Element and Locator array in order through the real console', () => {
  const host = create(),
    logs = [],
    original = console.log
  try {
    const locator = page.getByRole('button', { name: 'Save' })
    console.log = function (...args) {
      logs.push({ receiver: this === console, args })
      return Reflect.apply(original, this, args)
    }
    const returned = utils.debug([host, locator], 10000, { highlight: false })
    console.log = original
    expect(logs).toHaveLength(2)
    expect(logs.every((value) => value.receiver)).toBe(true)
    expect(returned).toBeUndefined()
    expect(logs.map((log) => log.args)).toEqual([
      [expected.full],
      [expected.button],
    ])
  } finally {
    console.log = original
    host.remove()
  }
})
