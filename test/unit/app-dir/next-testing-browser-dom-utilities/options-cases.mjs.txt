const base = {
  include: ['fixtures.test.mjs'],
  maxWorkers: 1,
  testTimeout: 5000,
  hookTimeout: 5000,
  includeTaskLocation: false,
}

function values() {
  return [
    { label: 'omitted', omitted: true },
    { label: 'undefined', value: undefined },
    { label: 'null', value: null },
    { label: 'aria', value: 'aria' },
    { label: 'html', value: 'html' },
    { label: 'all', value: 'all' },
    { label: 'false', value: false },
    { label: 'zero', value: 0 },
    { label: 'empty', value: '' },
    { label: 'unknown', value: 'future-format' },
    { label: 'array', value: [] },
    { label: 'object', value: {} },
  ]
}

function inheritance() {
  return [
    { label: 'inherited-default' },
    { label: 'inherited-true', extend: true },
    { label: 'independent-false', extend: false },
    { label: 'inherited-empty', extend: true, locators: {} },
    { label: 'inherited-null', extend: true, locators: { errorFormat: null } },
    { label: 'inherited-undefined', extend: true, locators: { errorFormat: undefined } },
    { label: 'inherited-locators-null', extend: true, locators: null },
    { label: 'inherited-locators-undefined', extend: true, locators: undefined },
    { label: 'inherited-html', extend: true, locators: { errorFormat: 'html' } },
  ]
}

function instances() {
  return [
    { label: 'locators-omitted' },
    { label: 'locators-undefined', locators: undefined },
    { label: 'locators-null', locators: null },
    { label: 'locators-empty', locators: {} },
    ...values().filter(value => !value.omitted).map(value => ({ label: 'value-' + value.label, locators: { errorFormat: value.value } })),
  ]
}

function browser(name, provider) {
  return {
    enabled: true,
    headless: true,
    provider: provider(),
    instances: [{ browser: 'chromium', name }],
  }
}

function independentProject(label, cacheRoot, provider, locators, hasLocators) {
  const settings = browser(label, provider)
  if (hasLocators) settings.locators = locators
  const project = {
    extends: false,
    cacheDir: cacheRoot + '/' + label,
    test: { ...base, name: label, provide: { formatCase: label }, browser: settings },
  }
  delete project.test.maxWorkers
  return project
}

export function buildFormatConfiguration(group, { cacheRoot, provider }) {
  const configuration = { cacheDir: cacheRoot + '/root', test: { maxWorkers: 1 } }
  if (group === 'normalization') {
    configuration.test.projects = values().map(value => independentProject(
      'normalization-' + value.label, cacheRoot, provider,
      { errorFormat: value.value }, !value.omitted
    ))
  } else if (group === 'inheritance') {
    configuration.test = {
      ...base,
      browser: { enabled: true, headless: true, provider: provider(), instances: [], locators: { errorFormat: 'aria' } },
      projects: inheritance().map(value => {
        const label = 'inheritance-' + value.label
        if (value.extend === false) return independentProject(label, cacheRoot, provider, undefined, false)
        const settings = { instances: [{ browser: 'chromium', name: label }] }
        if (Object.hasOwn(value, 'locators')) settings.locators = value.locators
        return {
          ...(Object.hasOwn(value, 'extend') ? { extends: value.extend } : {}),
          cacheDir: cacheRoot + '/' + label,
          test: { name: label, provide: { formatCase: label }, browser: settings },
        }
      }),
    }
  } else if (group === 'instances') {
    configuration.test.projects = instances().map(value => {
      const label = 'instances-' + value.label
      const project = independentProject(label, cacheRoot, provider, { errorFormat: 'aria' }, true)
      const instance = { browser: 'chromium', name: label }
      if (Object.hasOwn(value, 'locators')) instance.locators = value.locators
      project.test.browser.instances = [instance]
      return project
    })
  } else throw new Error('Unknown private option group: ' + group)
  return configuration
}

function describeValue(value) {
  if (value === undefined) return { type: 'undefined' }
  if (value === null) return { type: 'null', value: null }
  return { type: Array.isArray(value) ? 'array' : typeof value, value }
}

function describeLocators(value) {
  if (value === undefined || value === null) return describeValue(value)
  return { type: 'object', ownErrorFormat: Object.hasOwn(value, 'errorFormat'), ...(Object.hasOwn(value, 'errorFormat') ? { errorFormat: describeValue(value.errorFormat) } : {}) }
}

export function describeConfigurations() {
  return {
    normalization: values().map(({ label, omitted, value }) => ({ label: 'normalization-' + label, extends: false, locators: omitted ? { omitted: true } : { errorFormat: describeValue(value) } })),
    inheritance: inheritance().map(value => ({ label: 'inheritance-' + value.label, extends: Object.hasOwn(value, 'extend') ? value.extend : 'omitted', rootErrorFormat: 'aria', childLocators: Object.hasOwn(value, 'locators') ? describeLocators(value.locators) : { omitted: true } })),
    instances: instances().map(value => ({ label: 'instances-' + value.label, parentErrorFormat: 'aria', instanceLocators: Object.hasOwn(value, 'locators') ? describeLocators(value.locators) : { omitted: true } })),
  }
}
