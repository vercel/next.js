import { test, expect, inject } from 'vitest'
import { server, utils } from 'vitest/browser'

function emit(value) { console.log('BROWSER_NATIVE_FIXTURE:' + JSON.stringify(value)) }
function retainRawStack(value) { console.log('BROWSER_NATIVE_RAW_STACK:' + JSON.stringify(value)) }

test('formats a real DOM error with the resolved locator option', function namedConfiguredError() {
  const label = inject('formatCase')
  const host = document.createElement('section')
  host.innerHTML = '<h2>Hello</h2><button aria-label="Save">go</button>'
  document.body.append(host)
  try {
    utils.configurePrettyDOM({ highlight: false })
    const value = server.config.browser.locators.errorFormat
    const error = utils.getElementError('css=.missing', host)
    const stack = error.stack
    expect(error).toBeInstanceOf(Error)
    emit({
      label,
      errorFormat: { type: typeof value, array: Array.isArray(value), value },
      error: {
        instanceOfError: error instanceof Error,
        prototypeIsNativeError: Object.getPrototypeOf(error) === Error.prototype,
        constructorIsNativeError: error.constructor === Error,
        name: error.name,
        message: error.message,
        ownKeys: Object.getOwnPropertyNames(error),
        stackType: typeof stack,
      },
    })
    retainRawStack({ label, stack })
  } finally {
    utils.configurePrettyDOM(undefined)
    host.remove()
  }
})
