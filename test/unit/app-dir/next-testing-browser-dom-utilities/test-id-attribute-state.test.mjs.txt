import { test, expect } from 'vitest'
import { page, server, locators } from 'vitest/browser'

const selector = (attribute, id) => `internal:testid=[${attribute}="${id}"s]`
function assertTarget(locator, target, expectedSelector) {
  expect(locator.selector).toBe(expectedSelector)
  expect(locator.element()).toBe(target)
}
function withState(callback) {
  const publicDescriptor = Object.getOwnPropertyDescriptor(server, 'config')
  const initialConfig = server.config,
    initialLocators = initialConfig.browser.locators
  const attributeDescriptor = Object.getOwnPropertyDescriptor(
    initialLocators,
    'testIdAttribute'
  )
  expect(initialLocators.testIdAttribute).toBe('data-qa')
  const host = document.createElement('section')
  host.setAttribute('data-qa', 'group')
  host.innerHTML =
    '<button data-testid="legacy" data-qa="configured" data-next="changed" data-replacement="replaced" data-getter="getter"></button>'
  document.body.append(host)
  const target = host.firstElementChild
  const oldPage = page.getByTestId('configured')
  const base = page.elementLocator(host)
  const wrapper = locators.createElementLocators(host)
  const oldChild = base.getByTestId('configured')
  const oldWrapperChild = wrapper.getByTestId('configured')
  try {
    callback({
      initialConfig,
      initialLocators,
      host,
      target,
      oldPage,
      base,
      wrapper,
      oldChild,
      oldWrapperChild,
    })
  } finally {
    if (attributeDescriptor)
      Object.defineProperty(
        initialLocators,
        'testIdAttribute',
        attributeDescriptor
      )
    else delete initialLocators.testIdAttribute
    Object.defineProperty(server, 'config', publicDescriptor)
    host.remove()
    expect(server.config).toBe(initialConfig)
    expect(Object.getOwnPropertyDescriptor(server, 'config')).toEqual(
      publicDescriptor
    )
    expect(
      Object.getOwnPropertyDescriptor(initialLocators, 'testIdAttribute')
    ).toEqual(attributeDescriptor)
  }
}
const captured = selector('data-qa', 'configured')
const group = selector('data-qa', 'group')

test('initial custom attribute has public alias and captured selectors', () =>
  withState(
    ({
      initialConfig,
      initialLocators,
      target,
      oldPage,
      base,
      oldChild,
      oldWrapperChild,
    }) => {
      expect(server.config).toBe(initialConfig)
      expect(initialLocators.testIdAttribute).toBe('data-qa')
      assertTarget(oldPage, target, captured)
      assertTarget(oldChild, target, group + ' >> ' + captured)
      assertTarget(oldWrapperChild, target, group + ' >> ' + captured)
      assertTarget(page.elementLocator(target), target, captured)
      assertTarget(base.elementLocator(target), target, captured)
    }
  ))

test('nested mutation retargets new calls and retains old selectors and initial engine', () =>
  withState(
    ({
      initialConfig,
      initialLocators,
      target,
      oldPage,
      base,
      wrapper,
      oldChild,
    }) => {
      initialLocators.testIdAttribute = 'data-next'
      expect(server.config).toBe(initialConfig)
      expect(server.config.browser.locators.testIdAttribute).toBe('data-next')
      assertTarget(oldPage, target, captured)
      assertTarget(oldChild, target, group + ' >> ' + captured)
      const changed = selector('data-next', 'changed')
      assertTarget(page.getByTestId('changed'), target, changed)
      assertTarget(
        base.getByTestId('changed'),
        target,
        group + ' >> ' + changed
      )
      assertTarget(
        wrapper.getByTestId('changed'),
        target,
        group + ' >> ' + changed
      )
      assertTarget(page.elementLocator(target), target, captured)
      assertTarget(base.elementLocator(target), target, captured)
    }
  ))

test('public replacement retargets new calls while original alias and engine remain distinct', () =>
  withState(
    ({
      initialConfig,
      initialLocators,
      target,
      oldPage,
      base,
      wrapper,
      oldChild,
    }) => {
      const replacement = {
        ...initialConfig,
        browser: {
          ...initialConfig.browser,
          locators: { ...initialLocators, testIdAttribute: 'data-replacement' },
        },
      }
      server.config = replacement
      initialLocators.testIdAttribute = 'data-next'
      expect(server.config).toBe(replacement)
      expect(server.config).not.toBe(initialConfig)
      expect(initialLocators.testIdAttribute).toBe('data-next')
      expect(server.config.browser.locators.testIdAttribute).toBe(
        'data-replacement'
      )
      assertTarget(oldPage, target, captured)
      assertTarget(oldChild, target, group + ' >> ' + captured)
      const replaced = selector('data-replacement', 'replaced')
      assertTarget(page.getByTestId('replaced'), target, replaced)
      assertTarget(
        base.getByTestId('replaced'),
        target,
        group + ' >> ' + replaced
      )
      assertTarget(
        wrapper.getByTestId('replaced'),
        target,
        group + ' >> ' + replaced
      )
      assertTarget(page.elementLocator(target), target, captured)
      assertTarget(base.elementLocator(target), target, captured)
    }
  ))

test('getter chain is read by new calls and not captured selectors or initial engine', () =>
  withState(
    ({
      initialConfig,
      initialLocators,
      host,
      target,
      oldPage,
      base,
      wrapper,
    }) => {
      const calls = [],
        expected = ['server.config', 'browser', 'locators', 'testIdAttribute']
      const watchedLocators = { ...initialLocators }
      Object.defineProperty(watchedLocators, 'testIdAttribute', {
        configurable: true,
        enumerable: true,
        get() {
          calls.push('testIdAttribute')
          return 'data-getter'
        },
      })
      const watchedBrowser = { ...initialConfig.browser }
      Object.defineProperty(watchedBrowser, 'locators', {
        configurable: true,
        enumerable: true,
        get() {
          calls.push('locators')
          return watchedLocators
        },
      })
      const watchedConfig = { ...initialConfig }
      Object.defineProperty(watchedConfig, 'browser', {
        configurable: true,
        enumerable: true,
        get() {
          calls.push('browser')
          return watchedBrowser
        },
      })
      Object.defineProperty(server, 'config', {
        configurable: true,
        enumerable: true,
        get() {
          calls.push('server.config')
          return watchedConfig
        },
      })
      const observe = (create, reads, expectedSelector) => {
        calls.length = 0
        const locator = create()
        expect(calls).toEqual(reads)
        assertTarget(locator, target, expectedSelector)
        expect(calls).toEqual(reads)
      }
      const getter = selector('data-getter', 'getter')
      observe(() => oldPage, [], captured)
      observe(() => page.getByTestId('getter'), expected, getter)
      observe(
        () => base.getByTestId('getter'),
        expected,
        group + ' >> ' + getter
      )
      observe(
        () => wrapper.getByTestId('getter'),
        expected,
        group + ' >> ' + getter
      )
      observe(() => page.elementLocator(target), [], captured)
      observe(() => base.elementLocator(target), [], captured)
      calls.length = 0
      const freshWrapper = locators.createElementLocators(host)
      expect(calls).toEqual([])
      observe(
        () => freshWrapper.getByTestId('getter'),
        expected,
        group + ' >> ' + getter
      )
    }
  ))
