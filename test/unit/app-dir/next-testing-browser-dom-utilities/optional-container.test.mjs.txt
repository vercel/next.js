const captureNativeErrorFields = error => ({instanceOfError:error instanceof Error,constructor:error?.constructor?.name,name:error?.name,message:error?.message,stack:error?.stack,ownKeys:Reflect.ownKeys(error).map(String),descriptors:Object.fromEntries(Object.getOwnPropertyNames(error).map(key=>{const d=Object.getOwnPropertyDescriptor(error,key);return[key,{enumerable:d.enumerable,configurable:d.configurable,writable:d.writable}]}))})
const fields = error => {
 const captured = captureNativeErrorFields(error)
 const { stack, ...semantic } = captured
 console.log('BROWSER_NATIVE_STACK:' + JSON.stringify({name:captured.name,message:captured.message,stack}))
 return {...semantic,stackType:typeof stack}
}

const create = () => { const host=document.createElement('section');host.innerHTML='<h2>Hello</h2><button aria-label="Save">go</button>';document.body.append(host);return host }
for(const role of ['omitted','undefined','null','element'])test('native optional-container '+role,()=>{
 const host=create(),calls=[],selector={asLocator(){calls.push({kind:'selector',receiver:this===selector});return'authored locator'}}
 try{utils.configurePrettyDOM({highlight:false});const result=observe(()=>role==='omitted'?utils.getElementError(selector):utils.getElementError(selector,role==='element'?host:role==='null'?null:undefined));expect(result.error instanceof Error).toBe(true);emit({role,kind:result.kind,calls,error:fields(result.error)})}
 finally{utils.configurePrettyDOM(undefined);host.remove()}
})

 finally{utils.configurePrettyDOM(undefined)}
})
