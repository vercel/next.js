import { buildFormatConfiguration, describeConfigurations } from './options-cases.mjs'

export function describeSingleCases() {
  return Object.entries(describeConfigurations()).flatMap(([group, cases]) => cases.map(description => ({ group, id: description.label, description })))
}

export function buildSingleCaseConfiguration(id, options) {
  const entry = describeSingleCases().find(entry => entry.id === id)
  if (!entry) throw new Error('Unknown private single case: ' + id)
  const configuration = buildFormatConfiguration(entry.group, options)
  const project = configuration.test.projects.find(project => project.test.name === id)
  if (!project) throw new Error('The selected private project is missing: ' + id)
  configuration.test.projects = [project]
  return configuration
}
