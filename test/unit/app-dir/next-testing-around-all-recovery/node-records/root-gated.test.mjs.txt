import { aroundAll, beforeAll, afterAll, describe, expect, test } from 'vitest'
import { appendFileSync } from 'node:fs'
const phase = process.env.AROUND_PHASE
const followup = process.env.AROUND_FOLLOWUP
let releaseSettled
const settled = new Promise(resolve => { releaseSettled = resolve })
const counts = { affectedSetup: 0, affectedBody: 0, affectedCleanup: 0, wrapper: 0, followingSetup: 0, followingBody: 0, followingCleanup: 0, rootCleanup: 0 }
const row = (event, extra = {}) => appendFileSync(process.env.AROUND_TRACE, JSON.stringify({ event, phase, pid: process.pid, now: Date.now(), counts: { ...counts }, ...extra }) + '\n')
const errorFields = error => ({ name: error?.name, message: error?.message, stack: error?.stack })
describe('affected', () => {
  aroundAll(async run => {
    counts.wrapper++
    row('wrapper-entry')
    try {
      if (phase === 'teardown') {
        row('run-before')
        await run()
        row('run-after')
      }
      row('bounded-wait-entry')
      await new Promise(resolve => setTimeout(resolve, 1000))
      row('bounded-wait-settled')
      if (phase === 'setup') {
        row('late-run-before')
        await run()
        row('late-run-after')
      }
    } catch (error) {
      row('wrapper-caught-rethrown', { error: errorFields(error) })
      throw error
    } finally {
      row('wrapper-finally')
      releaseSettled()
    }
  }, 100)
  beforeAll(() => { counts.affectedSetup++; row('affected-beforeAll') })
  afterAll(() => { counts.affectedCleanup++; row('affected-afterAll') })
  test('body', () => {
    counts.affectedBody++
    row('affected-body-before', { assertions: expect.getState().assertionCalls })
    expect(true).toBe(true)
    row('affected-body-after', { assertions: expect.getState().assertionCalls })
  })
})
describe('following', () => {
  beforeAll(async () => {
    counts.followingSetup++
    row('following-beforeAll-entry')
    row('following-beforeAll-returned')
  })
  afterAll(() => { counts.followingCleanup++; row('following-afterAll') })
  test('body', () => {
    counts.followingBody++
    row('following-body-before', { assertions: expect.getState().assertionCalls })
    expect(true).toBe(true)
    row('following-body-after', { assertions: expect.getState().assertionCalls })
  })
})

afterAll(async () => {
  counts.rootCleanup++
  row('root-afterAll-entry')
  if (followup === 'root-gated') {
    await settled
    row('root-afterAll-settled')
  }
  row('root-afterAll-returned')
})
