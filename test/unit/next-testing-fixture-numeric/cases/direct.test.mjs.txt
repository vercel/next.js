import { test, describe, expect } from 'vitest'
const row = (phase) =>
  console.log(
    'FIXTURE_DEPENDENCY_ROW ' + JSON.stringify({ contract: 'direct', phase })
  )
const owned = test.extend({
  42: async ({}, use) => {
    row('setup')
    await use(42)
    row('cleanup')
  },
})
owned('numeric direct', ({ 42: value }) => {
  row('body', value)
  expect(value).toBe(42)
})
