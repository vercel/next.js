import { test, describe, expect } from 'vitest'
const row = (phase) =>
  console.log(
    'FIXTURE_DEPENDENCY_ROW ' + JSON.stringify({ contract: 'chain', phase })
  )
const owned = test.extend({
  42: async ({}, use) => {
    row('setup')
    await use(42)
    row('cleanup')
  },
  answer: async ({ 42: value }, use) => {
    row('answer setup', value)
    await use(value + 1)
    row('answer cleanup')
  },
})
owned('numeric dependency', ({ answer }) => {
  row('body', answer)
  expect(answer).toBe(43)
})
