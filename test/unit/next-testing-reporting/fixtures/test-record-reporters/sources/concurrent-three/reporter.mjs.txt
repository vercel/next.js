import fs from 'node:fs'
import path from 'node:path'
const label = "A", mode = "concurrent"
const record = row => fs.appendFileSync(process.env.AA_TRACE, JSON.stringify({ source: label, ...row }) + '\n')
const error = name => Object.assign(new Error('authored reporter ' + name), { marker: 'reporter-marker', cause: new Error('reporter-cause') })
const cases = new Map()
const gates = []
const primitive = kind => kind === 'undefined' ? undefined : kind === 'null' ? null : kind === 'false' ? false : kind === 'zero' ? 0 : ''
const entity = test => ({ type: test.type, id: test.id, name: test.name, fullName: test.fullName, project: test.project.name, parent: { type: test.parent?.type, name: test.parent?.name }, module: { moduleId: test.module.moduleId, relativeModuleId: test.module.relativeModuleId }, options: test.options, result: test.result(), annotations: test.annotations(), artifacts: test.artifacts(), meta: test.meta() })
const factory = (self, name) => ({ configurable: true, get() {
  record({ event: 'getter', name, receiver: this === self })
  if (mode === 'undefined') return undefined
  if (mode === 'getter-throw-annotation' && name === 'onTestCaseAnnotate') throw error('getter')
  return function (...args) {
    const [test, value] = args
    record({ event: 'callback-enter', name, argc: args.length, receiver: this === self, sameCaseReady: cases.get(test.id) === test, sameArrayValue: (name === 'onTestCaseAnnotate' ? test.annotations() : test.artifacts()).at(-1) === value, value, entity: entity(test) })
    const finish = () => {
      if (mode.startsWith('primitive-')) throw primitive(mode.split('-').at(-1))
      if (mode === 'mutate') { value.reporterAdded = label; if (name === 'onTestCaseAnnotate') value.message += '-changed'; else { value.title += '-changed'; value.data.counter++ } }
      if (mode === 'sync-throw-annotation' && name === 'onTestCaseAnnotate') throw error('sync')
      if ((mode === 'async-reject-annotation' && name === 'onTestCaseAnnotate') || (mode === 'async-reject-artifact' && name === 'onTestCaseArtifactRecord') || mode === 'reject-both') throw error('async')
      record({ event: 'callback-exit', name, value, entity: entity(test) })
      return { ignored: 'reporter-return' }
    }
    if (mode === 'concurrent') return new Promise(resolve => { gates.push({ name, value, release: resolve }); record({ event: 'gate-enter', name, slot: gates.length, message: value.message ?? value.title }); if (gates.length === 3) { for (const gate of [gates.find(g => g.name === 'onTestCaseArtifactRecord'), gates.find(g => g.value.message === 'parallel-two'), gates.find(g => g.value.message === 'parallel-one')]) gate.release() } }).then(() => { record({ event: 'gate-release', name, message: value.message ?? value.title }); value.reporterAdded = 'concurrent'; return finish() })
    if (mode === 'async' || mode.startsWith('async-reject-') || mode === 'reject-both' || mode.startsWith('primitive-async-')) return (async () => { record({ event: 'async-prefix', name }); await new Promise(resolve => setImmediate(resolve)); return finish() })()
    return finish()
  }
} })
export default class {
  constructor() { for (const name of ['onTestCaseAnnotate', 'onTestCaseArtifactRecord']) Object.defineProperty(this, name, factory(this, name)) }
  onInit(context) { record({ event: 'init' }); context.onClose(() => record({ event: 'close' })) }
  onTestCaseReady(test) { cases.set(test.id, test); record({ event: 'case-ready', entity: entity(test) }) }
  onTestCaseResult(test) { record({ event: 'case-result', entity: entity(test) }) }
  onTestRunEnd(modules, errors, reason) { record({ event: 'run-end', reason, errors, modules: modules.map(m => ({ moduleId: m.moduleId, state: m.state(), errors: m.errors() })) }) }
}
