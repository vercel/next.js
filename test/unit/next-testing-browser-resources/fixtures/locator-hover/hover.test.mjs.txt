afterEach(() => { if (root) { flushSync(() => root.unmount()); root=undefined } document.body.replaceChildren(); document.body.removeAttribute('style') })
function settled(operation){try{return Promise.resolve(operation()).then(value=>({returnedUndefined:value===undefined}),error=>({name:error.name,message:error.message}))}catch(error){return Promise.resolve({name:error.name,message:error.message})}}
function trace(element) { const events=[]; for(const type of ['pointerover','pointerenter','mouseover','mouseenter','pointermove','mousemove','pointerout','pointerleave','mouseout','mouseleave']) element.addEventListener(type,event=>events.push({type:event.type,constructor:event.constructor.name,target:event.target===element,currentTarget:event.currentTarget===element,relatedBody:event.relatedTarget===document.body,trusted:event.isTrusted,bubbles:event.bubbles,composed:event.composed,detail:event.detail,button:event.button,buttons:event.buttons,x:event.clientX,y:event.clientY,text:element.textContent}));return events }
async function anchor(){document.body.style.cssText='margin:0;min-height:480px;position:relative';const button=document.createElement('button');button.textContent='Pointer anchor';button.style.cssText='position:absolute;left:20px;top:20px;width:80px;height:30px';document.body.append(button);await page.getByRole('button',{name:'Pointer anchor',exact:true}).click();return {element:button,events:trace(button)}}
