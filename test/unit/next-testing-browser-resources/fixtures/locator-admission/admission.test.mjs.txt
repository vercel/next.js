import { afterEach, expect, test } from 'vitest'
import { page } from 'vitest/browser'
import { createElement, useState } from 'react'
import { createRoot } from 'react-dom/client'
import { flushSync } from 'react-dom'
let reactRoot
afterEach(() => { if (reactRoot) { flushSync(() => reactRoot.unmount()); reactRoot = undefined } document.body.replaceChildren() })
function trace(element) { const events = []; for (const type of ['pointerdown', 'mousedown', 'pointerup', 'mouseup', 'click', 'input', 'change']) element.addEventListener(type, event => events.push({ type, constructor: event.constructor.name, target: event.target === element, currentTarget: event.currentTarget === element, trusted: event.isTrusted, bubbles: event.bubbles, composed: event.composed, detail: event.detail, button: event.button, buttons: event.buttons, value: element.value, text: element.textContent })); return events }
function record(name, element, events, outcome) { console.log('LOCATOR_ADMISSION_RAW ' + JSON.stringify({ name, mode: process.env.NODE_ENV, connected: element.isConnected, value: element.value, text: element.textContent, events, outcome })) }
function settled(promise) { return promise.then(value => ({ returnedUndefined: value === undefined }), error => ({ errorName: error.name, errorMessage: error.message })) }
test('click waits for a role query created before the button exists', async () => {
  const call = settled(page.getByRole('button', { name: 'Late button', exact: true }).click({ timeout: 500 }))
  await Promise.resolve()
  const button = document.createElement('button'); button.textContent = 'Late button'; const events = trace(button); document.body.append(button)
  const outcome = await call; record('late DOM click', button, events, outcome)
  expect(outcome).toEqual({ returnedUndefined: true }); expect(events.map(event => event.type)).toContain('click')
})
test('fill waits for a label query created before the input exists', async () => {
  const call = settled(page.getByLabelText('Late input').fill('native value', { timeout: 500 }))
  await Promise.resolve()
  const input = document.createElement('input'); input.setAttribute('aria-label', 'Late input'); const events = trace(input); document.body.append(input)
  const outcome = await call; record('late DOM fill', input, events, outcome)
  expect(outcome).toEqual({ returnedUndefined: true }); expect(input.value).toBe('native value')
})
test('click waits for a real React button and its native render turn', async () => {
  function Controlled() { const [clicked, setClicked] = useState(false); return createElement('button', { onClick: () => setClicked(true) }, clicked ? 'React clicked' : 'Late React') }
  const call = settled(page.getByRole('button', { name: 'Late React', exact: true }).click({ timeout: 500 }))
  await Promise.resolve()
  const container = document.createElement('div'); document.body.append(container); reactRoot = createRoot(container); flushSync(() => reactRoot.render(createElement(Controlled))); const button = container.querySelector('button'), events = trace(button)
  const outcome = await call; record('late React click', button, events, outcome)
  expect(outcome).toEqual({ returnedUndefined: true }); expect(button.textContent).toBe('React clicked')
})
test('an existing disabled button retains native actionability waiting', async () => {
  const button = document.createElement('button'); button.textContent = 'Disabled control'; button.disabled = true; document.body.append(button); const events = trace(button)
  const call = settled(page.getByRole('button', { name: 'Disabled control', exact: true }).click({ timeout: 500 }))
  await Promise.resolve(); button.disabled = false
  const outcome = await call; record('disabled native control', button, events, outcome)
  expect(outcome).toEqual({ returnedUndefined: true }); expect(events.map(event => event.type)).toContain('click')
})
