import { expect, test } from 'vitest'

test('uses the Browser viewport and default document geometry', () => {
  const marker = document.createElement('div')
  marker.style.width = '100px'
  marker.style.height = '50px'
  document.body.append(marker)
  try {
    expect(marker.isConnected).toBe(true)
    expect(getComputedStyle(document.documentElement).margin).toBe('0px')
    expect(getComputedStyle(document.body).margin).toBe('0px')
    expect(getComputedStyle(document.body).padding).toBe('0px')
    const { left, top, width, height } = marker.getBoundingClientRect()
    expect({ left, top, width, height }).toEqual({
      left: 0,
      top: 0,
      width: 100,
      height: 50,
    })
    expect([innerWidth, innerHeight]).toEqual([640, 480])
  } finally {
    marker.remove()
  }
})
