import assert from 'node:assert/strict'
import { afterAll, expect, test } from 'vitest'
let held,
  caller,
  taskA,
  taskB,
  expectedTask,
  expectedAssertion,
  reads = 0,
  phases = 0
const sized = {
  get length() {
    reads++
    return 2
  },
}
const operations = [
  ['toBeTypeOf', 2, ['number']],
  ['toBeDefined', 1, []],
  ['toBeUndefined', undefined, []],
  ['toBeNull', null, []],
  ['toBeNaN', NaN, []],
  ['toBeTruthy', 1, []],
  ['toBeFalsy', 0, []],
  ['toBeGreaterThan', 3, [2]],
  ['toBeGreaterThanOrEqual', 2, [2]],
  ['toBeLessThan', 1, [2]],
  ['toBeLessThanOrEqual', 2, [2]],
  ['toBeCloseTo', 0.1 + 0.2, [0.3, 5]],
  ['toBeInstanceOf', new Map(), [Map]],
  ['toContain', [1, 2], [2]],
  ['toContainEqual', [{ a: 1 }], [{ a: 1 }]],
  ['toHaveLength', sized, [2]],
  ['toMatch', 'hello actual task', [/actual/]],
]
const captures = []
expect.extend({
  toInspectBuiltinOwner() {
    assert.equal(this.task, expectedTask)
    assert.equal(this.assertion, expectedAssertion)
    return { pass: true, message: () => 'hard receiver changed' }
  },
})
function checkCall(method, receiver, args, task) {
  const returned = Reflect.apply(method, receiver, args)
  assert.equal(typeof returned, 'object')
  assert.notEqual(returned, receiver)
  expectedTask = task
  expectedAssertion = returned
  returned.toInspectBuiltinOwner()
}
function run() {
  phases++
  for (const [, value, args, assertion, method] of captures) {
    checkCall(method, assertion, args, taskA)
    assert.throws(
      () => Reflect.apply(method.bind(assertion), undefined, args),
      /Invalid Chai property/
    )
    checkCall(method, caller(value), args, taskB)
    assert.throws(() => Reflect.apply(method, undefined, args), TypeError)
    const raw = {}
    assert.throws(() => Reflect.apply(method, raw, args), TypeError)
  }
  assert.equal(held.getState().assertionCalls, 17)
  assert.equal(caller.getState().assertionCalls, phases * 17)
  assert.equal(reads, phases * 4)
}
test('creator hard receivers', (context) => {
  held = context.expect
  taskA = context.task
  for (const [name, value, args] of operations) {
    const assertion = held(value)
    captures.push([name, value, args, assertion, assertion[name]])
  }
})
test('foreign hard receivers', (context) => {
  caller = context.expect
  taskB = context.task
  run()
})
afterAll(() => {
  run()
  console.log(
    'DECLARATION_FN:' +
      JSON.stringify({
        kind: 'late-hard-builtin-receivers',
        operations: operations.length,
        phases,
      })
  )
})
