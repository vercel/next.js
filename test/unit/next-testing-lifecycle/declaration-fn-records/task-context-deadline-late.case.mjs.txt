import {describe,afterAll} from 'vitest';
const flags={concurrent:false,only:false,skip:false,todo:false,shuffle:false},trace=[],held=[],turns=new Map(),blocks=new Map();
function add(name,action){let task;task=collector.task(name,{retry:1,timeout:60,async handler(context){const turn=turns.get(name)??0;turns.set(name,turn+1);trace.push({kind:'body',name,turn,count:context.expect.getState().assertionCalls,aborted:context.signal.aborted});context.onTestFinished(finished=>trace.push({kind:'finished',name,turn,count:finished.expect.getState().assertionCalls,state:finished.task.result?.state??null}));if(turn===0){let release,joined;const gate=new Promise(resolve=>release=resolve),join=new Promise(resolve=>joined=resolve);blocks.set(name,{release,join});await gate;trace.push({kind:'late-start',name,state:context.task.result?.state??null});try{action(context);trace.push({kind:'late-action-ok',name});}finally{joined();}return;}const block=blocks.get(name);block.release();await block.join;await Promise.resolve();context.expect(true).toBe(true);trace.push({kind:'retry-end',name,count:context.expect.getState().assertionCalls});}});held.push(task);}
const collector=describe.fn.call(flags,'late',{},()=>{});
add('old-rejection',()=>{throw new Error('old body rejection after retry release');});
add('old-local-assertion',context=>context.expect('old continuation').toBe('old continuation'));
afterAll(async()=>{for(const block of blocks.values()){block.release();await block.join;}await Promise.resolve();console.log('DECLARATION_FN:'+JSON.stringify({kind:'task-context-deadline-late',trace,tasks:held.map(task=>({name:task.name,state:task.result?.state??null,errors:task.result?.errors?.map(error=>({name:error.name,message:error.message}))??[]}))}));});
