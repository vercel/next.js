import {describe,afterAll} from 'vitest';
const flags={concurrent:false,only:false,skip:false,todo:false,shuffle:false},trace=[],held=[],turns=new Map();
const reason=signal=>signal.reason?{name:signal.reason.name,message:signal.reason.message}:null;
let releaseDeadline,joinDeadline;const deadlineGate=new Promise(resolve=>{releaseDeadline=resolve}),deadlineJoined=new Promise(resolve=>{joinDeadline=resolve});
function add(owner,name,options,body){let task;task=owner.task(name,{...options,async handler(context){const turn=turns.get(name)??0;turns.set(name,turn+1);const saved=held.find(value=>value.name===name);trace.push({kind:'body',name,turn,retry:task.result?.retryCount??0,repeat:task.result?.repeatCount??0,sameContext:context===saved.context,taskIdentity:context.task===task,signalSame:context.signal===saved.signal,expectSame:context.expect===saved.localExpect,signalAborted:context.signal.aborted,signalReason:reason(context.signal),assertionCalls:context.expect.getState().assertionCalls});context.onTestFinished(finished=>{trace.push({kind:'finished',name,turn,sameContext:finished===saved.context,signalSame:finished.signal===saved.signal,signalAborted:finished.signal.aborted,signalReason:reason(finished.signal),assertionCalls:finished.expect.getState().assertionCalls});});await body(context,turn);}});const context=task.context,signal=context.signal,localExpect=context.expect;held.push({name,task,context,signal,localExpect});localExpect(true).toBe(true);trace.push({kind:'collection',name,contextType:typeof context,signalAborted:signal.aborted,assertionCalls:localExpect.getState().assertionCalls});}
const collector=describe.fn.call(flags,'signals',{},()=>add(collector,'repeated',{repeats:1},context=>context.expect(true).toBe(true)));
add(collector,'retried',{retry:1},(context,turn)=>context.expect(turn).toBe(1));
add(collector,'deadline',{retry:1,timeout:100},async(context,turn)=>{if(turn===0){await deadlineGate;trace.push({kind:'late-body',name:'deadline',signalAborted:context.signal.aborted,signalReason:reason(context.signal)});joinDeadline();return;}releaseDeadline();await deadlineJoined;context.expect(true).toBe(true);});
add(collector,'after-deadline',{},context=>{const previous=held.find(value=>value.name==='deadline');context.expect(previous.signal.aborted).toBe(true);context.expect(context.signal===previous.signal).toBe(false);});
afterAll(()=>console.log('DECLARATION_FN:'+JSON.stringify({kind:'task-context-signals',trace,contexts:held.map(({name,task,context,signal,localExpect})=>({name,contextType:typeof context,sameContext:task.context===context,signalAborted:signal.aborted,signalReason:reason(signal),assertionCalls:localExpect.getState().assertionCalls,state:task.result?.state??null}))})));
