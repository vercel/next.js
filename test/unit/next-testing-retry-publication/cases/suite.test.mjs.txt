import { describe, test, expect } from 'vitest'
const row = (...event) => console.log('RETRY_PUBLICATION_ROW ' + JSON.stringify({ actor: 'worker', contract: 'suite-inheritance', event }))
const inheritedMetadata = { value: 42 }
const inherited = { count: 0, metadata: inheritedMetadata }
const ownMetadata = [7]
const own = { count: 0, metadata: ownMetadata }
describe('inherited suite retry', { retry: inherited }, () => {
  test('inherits suite retry', ({ task }) => { row('inherited', task.retry === inherited, inherited.metadata === inheritedMetadata); expect(task.retry).toBe(inherited) })
  test('replaces suite retry', { retry: own }, ({ task }) => { row('own', task.retry === own, own.metadata === ownMetadata); expect(task.retry).toBe(own) })
})
row('registered')
